```python
# (1) Tách văn bản thành tập hợp các 3-gram từ (word shingles)
def get_shingles(text, n=3):
    clean = "".join(c.lower() for c in text if c.isalnum() or c == " ")
    words = clean.split()
    return set(" ".join(words[i : i + n]) for i in range(len(words) - n + 1))


# (2) Tính chữ ký MinHash với K = 64 hàm băm độc lập
def compute_minhash(shingles, num_perm=64):
    sig = []
    for seed in range(num_perm):
        min_h = min(
            int.from_bytes(
                hashlib.md5(f"{seed}:{s}".encode()).digest()[:4], "big"
            )
            for s in shingles
        )
        sig.append(min_h)
    return sig


shingle_sets = [get_shingles(doc, n=3) for doc in corpus]
signatures = [compute_minhash(s, num_perm=64) for s in shingle_sets]

# (3) So sánh các cặp và lọc bỏ tài liệu trùng lặp (ngưỡng >= 0.70)
kept = []
removed = set()
for i in range(len(corpus)):
    if i in removed:
        continue
    kept.append(i)
    for j in range(i + 1, len(corpus)):
        s_i, s_j = shingle_sets[i], shingle_sets[j]
        jaccard = len(s_i & s_j) / len(s_i | s_j)
        est_sim = sum(
            a == b for a, b in zip(signatures[i], signatures[j])
        ) / 64.0
        if est_sim >= 0.70:
            removed.add(j)
            print(f"Cặp ({i}, {j}): Jaccard thực = {jaccard:.3f} | "
                  f"MinHash ước lượng = {est_sim:.3f} -> TRÙNG LẶP")

print(f"\nSố tài liệu sau khi khử trùng lặp (ngưỡng 0.70): "
      f"{len(kept)} / {len(corpus)}")
print(f"Giữ lại các tài liệu có chỉ số: {kept}")
```
