from datasets import load_dataset

imdb = load_dataset("stanfordnlp/imdb")


# (1) Tách 5.000 bài đánh giá huấn luyện làm tập xác thực, và
# thay thế các thẻ xuống dòng HTML bằng khoảng trắng
# [Sao chép từ lời giải bài 3.2, bước (1)]

def clean(texts):
    return [text.replace("<br />", " ") for text in texts]

split = imdb["train"].train_test_split(test_size=5000, seed=0)
train_texts = clean(split["train"]["text"])
val_texts = clean(split["test"]["text"])
test_texts = clean(imdb["test"]["text"])
train_labels = split["train"]["label"]
val_labels = split["test"]["label"]
test_labels = imdb["test"]["label"]

print(f"train: {len(train_texts)} reviews")
print(f"validation: {len(val_texts)} reviews")
print(f"test: {len(test_texts)} reviews")


# (2) Đếm từ trong mỗi bài đánh giá trên 10.000 từ phổ biến nhất
# của tập huấn luyện. Biểu thức chính quy coi mọi chuỗi chữ cái
# hoặc chữ số là một từ, kể cả từ chỉ gồm một chữ cái như "a".
# [Sao chép từ lời giải bài 3.2, bước (2)]
from sklearn.feature_extraction.text import CountVectorizer

vectorizer = CountVectorizer(max_features=10000,
                             token_pattern=r"\b\w+\b")
X_train = vectorizer.fit_transform(train_texts)
X_val = vectorizer.transform(val_texts)
X_test = vectorizer.transform(test_texts)


# -------------------------

# được chọn trên tập xác thực
from sklearn.linear_model import LogisticRegression

model = LogisticRegression(C=0.1, max_iter=1000)
model.fit(X_train, train_labels)

# (1) Đếm các từ của những bài đánh giá khó trên từ vựng của
# mô hình, và tính xác suất để chúng thuộc lớp tích cực
hard_texts = [review for review, label in hard_reviews]
X_hard = vectorizer.transform(hard_texts)
probabilities = model.predict_proba(X_hard)
p_positive = probabilities[:, 1]   # cột 0: tiêu cực, 1: tích cực


# (2) In bảng kết quả: mô hình đoán đúng khi gán xác suất lớn hơn
# 0.5 cho nhãn thực tế của bài đánh giá
names = ["negative", "positive"]
print("label     p(positive)  right  review")
right_count = 0
for (review, label), p in zip(hard_reviews, p_positive):
    right = (p > 0.5) == (label == 1)
    if right:
        right_count += 1
    mark = "yes" if right else "NO"
    short = review if len(review) <= 36 else review[:35] + "…"
    print(f"{names[label]:9} {p:.3f}        {mark:5}  {short}")
print(f"\n{right_count} right out of {len(hard_reviews)}")
