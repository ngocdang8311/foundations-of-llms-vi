wikitext = load_dataset("Salesforce/wikitext", "wikitext-2-v1")


# (1) Build the train sequence by splitting every line on spaces

    train_sequence.extend(line.split())

print(f"vocabulary: {len(set(train_sequence))} words")


# The validation and test sequences, built in the same way

val_sequence = []
for line in wikitext["validation"]["text"]:
    val_sequence.extend(line.split())

test_sequence = []
for line in wikitext["test"]["text"]:
    test_sequence.extend(line.split())

# Give each word of the vocabulary an index
vocabulary = sorted(set(train_sequence))
word_index = {word: i for i, word in enumerate(vocabulary)}

# Turn a sequence into (context, next word) pairs of indices:
# row i of contexts holds the 3 words before the word targets[i]
import torch

context_size = 3

def make_pairs(sequence):
    indices = [word_index[word] for word in sequence]
    contexts = [indices[i - context_size:i]
                for i in range(context_size, len(indices))]
    targets = indices[context_size:]
    return torch.tensor(contexts), torch.tensor(targets)

train_contexts, train_targets = make_pairs(train_sequence)
val_contexts, val_targets = make_pairs(val_sequence)
test_contexts, test_targets = make_pairs(test_sequence)
