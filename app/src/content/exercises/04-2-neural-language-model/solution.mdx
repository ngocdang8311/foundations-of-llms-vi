from datasets import load_dataset

wikitext = load_dataset("Salesforce/wikitext", "wikitext-2-v1")


# (1) Tạo chuỗi huấn luyện bằng cách tách từ ở mỗi dòng theo dấu cách
# [Sao chép từ lời giải bài 2.1, bước (1)]
train_sequence = []
for line in wikitext["train"]["text"]:
    train_sequence.extend(line.split())

print(f"text: {len(train_sequence)} words")
print(f"vocabulary: {len(set(train_sequence))} words")


# Chuỗi xác thực (validation) và kiểm tra (test), tạo tương tự
# [Sao chép từ mã khởi đầu bài 2.2]
val_sequence = []
for line in wikitext["validation"]["text"]:
    val_sequence.extend(line.split())

test_sequence = []
for line in wikitext["test"]["text"]:
    test_sequence.extend(line.split())

print(f"validation: {len(val_sequence)} words")
print(f"test: {len(test_sequence)} words")


# -------------------------
#  Phần mới trong bài này
# -------------------------
# Gán một chỉ số (index) cho mỗi từ trong từ vựng
vocabulary = sorted(set(train_sequence))
word_index = {word: i for i, word in enumerate(vocabulary)}

# Chuyển chuỗi từ thành các cặp chỉ số (ngữ cảnh, từ tiếp theo):
# hàng i của contexts chứa 3 từ đứng trước từ targets[i]
import torch

context_size = 3

def make_pairs(sequence):
    indices = [word_index[word] for word in sequence]
    contexts = [indices[i - context_size:i]
                for i in range(context_size, len(indices))]
    targets = indices[context_size:]
    return torch.tensor(contexts), torch.tensor(targets)

train_contexts, train_targets = make_pairs(train_sequence)
val_contexts, val_targets = make_pairs(val_sequence)
test_contexts, test_targets = make_pairs(test_sequence)

print(f"train: {len(train_targets)} pairs")
context = [vocabulary[i] for i in train_contexts[30].tolist()]
print(f"example: {context} -> {vocabulary[train_targets[30]]}")

# (1) Xây dựng mô hình: tra cứu vector nhúng của 3 từ ngữ cảnh,
# ghép nối chúng lại, đi qua tầng ẩn (tanh) và xuất ra một điểm
# cho mỗi từ trong từ vựng. Khởi tạo các vector nhúng nhỏ: nếu
# dùng mặc định (cỡ 1), các từ hiếm sẽ giữ nguyên vector ngẫu nhiên.
from torch import nn

V = len(vocabulary)
D = 100   # kích thước vector nhúng
H = 200   # kích thước tầng ẩn
torch.manual_seed(0)
model = nn.Sequential(
    nn.Embedding(V, D),   # (batch, 3) -> (batch, 3, D)
    nn.Flatten(),         # ghép nối: (batch, 3 * D)
    nn.Linear(context_size * D, H),
    nn.Tanh(),
    nn.Linear(H, V),      # mỗi từ một điểm số
)
nn.init.normal_(model[0].weight, std=0.1)

# Chạy trên GPU nếu có
device = "cuda" if torch.cuda.is_available() else "cpu"
model = model.to(device)


# (2) Độ bối rối của mô hình trên tập các cặp từ: hàm mũ của
# hàm mất mát entropy chéo trung bình. F.cross_entropy áp dụng
# softmax lên điểm số rồi tính độ mất mát như ở mục 2.2.
import math
import torch.nn.functional as F

def perplexity(model, contexts, targets):
    total_loss = 0
    with torch.no_grad():
        for start in range(0, len(targets), 10000):
            scores = model(contexts[start:start + 10000].to(device))
            batch_targets = targets[start:start + 10000].to(device)
            total_loss += F.cross_entropy(scores, batch_targets,
                                          reduction="sum").item()
    return math.exp(total_loss / len(targets))


# (3) Huấn luyện bằng Adam trên từng lô 512 cặp. Sau mỗi epoch,
# đo độ bối rối trên tập xác thực và lưu bản sao mô hình tốt nhất.
import copy

optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
batch_size = 512
best_perplexity = math.inf
for epoch in range(4):
    order = torch.randperm(len(train_targets))   # xáo trộn các cặp
    for start in range(0, len(order), batch_size):
        batch = order[start:start + batch_size]
        scores = model(train_contexts[batch].to(device))
        batch_targets = train_targets[batch].to(device)
        loss = F.cross_entropy(scores, batch_targets)
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
    val_perplexity = perplexity(model, val_contexts, val_targets)
    print(f"epoch {epoch + 1}: validation perplexity "
          f"{val_perplexity:.1f}")
    if val_perplexity < best_perplexity:
        best_perplexity, best_epoch = val_perplexity, epoch + 1
        best_model = copy.deepcopy(model)


# (4) Độ bối rối trên tập kiểm tra của mô hình tốt nhất
test_perplexity = perplexity(best_model, test_contexts, test_targets)
print(f"\ntest perplexity, epoch {best_epoch}: {test_perplexity:.1f}")


# (5) Các láng giềng gần nhất của một vài từ. nn.Embedding lưu W
# ở dạng chuyển vị: hàng i là vector nhúng của từ i. Chia mỗi hàng
# cho độ dài của nó để tích vô hướng trở thành độ tương đồng cosine.
embeddings = best_model[0].weight.detach().cpu()
embeddings = embeddings / embeddings.norm(dim=1, keepdim=True)

print()
for word in ["three", "he", "city", "film", "excellent", "terrible"]:
    similarities = embeddings @ embeddings[word_index[word]]
    ranking = similarities.argsort(descending=True).tolist()
    nearest = [vocabulary[i] for i in ranking[1:6]]   # [0]: chính từ đó
    print(f"{word}: {', '.join(nearest)}")
