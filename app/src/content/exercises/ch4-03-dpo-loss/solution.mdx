# (1) Hàm tính tổng log-xác suất của chuỗi token y dưới phân phối logits
def sequence_logprob(logits: torch.Tensor, tokens: torch.Tensor) -> torch.Tensor:
    log_probs = F.log_softmax(logits, dim=-1)
    token_lp = log_probs.gather(dim=-1, index=tokens.unsqueeze(-1)).squeeze(-1)
    return token_lp.sum(dim=-1)


# (2) Hàm tính DPO Loss, phần thưởng ẩn r_hat và trọng số đạo hàm biên
def dpo_loss(pi_w, pi_l, ref_w, ref_l, y_w, y_l, beta=0.1):
    lp_pi_w = sequence_logprob(pi_w, y_w)
    lp_pi_l = sequence_logprob(pi_l, y_l)
    with torch.no_grad():
        lp_ref_w = sequence_logprob(ref_w, y_w)
        lp_ref_l = sequence_logprob(ref_l, y_l)

    r_w = beta * (lp_pi_w - lp_ref_w)
    r_l = beta * (lp_pi_l - lp_ref_l)
    margins = r_w - r_l

    loss = -F.logsigmoid(margins).mean()
    grad_weight = torch.sigmoid(-margins).mean()
    acc = (margins > 0).float().mean()
    return loss, margins.mean(), grad_weight, acc


init_loss, init_margin, _, init_acc = dpo_loss(
    pi_logits_w, pi_logits_l, ref_logits_w, ref_logits_l, y_w, y_l, beta
)
print(f"Loss DPO lúc khởi tạo (pi_theta = pi_ref): {init_loss.item():.6f} (ln 2 = 0.693147)")
print(f"Biên phần thưởng ẩn ban đầu: {init_margin.item():.4f} | Độ chính xác: {100*init_acc.item():.1f}%")

# (3) Tối ưu hóa trực tiếp logits của pi_theta bằng DPO
opt = torch.optim.SGD([pi_logits_w, pi_logits_l], lr=5.0)
for _ in range(10):
    opt.zero_grad()
    loss, margin, gw, acc = dpo_loss(
        pi_logits_w, pi_logits_l, ref_logits_w, ref_logits_l, y_w, y_l, beta
    )
    loss.backward()
    opt.step()

loss, margin, gw, acc = dpo_loss(
    pi_logits_w, pi_logits_l, ref_logits_w, ref_logits_l, y_w, y_l, beta
)
print("Sau 10 bước cập nhật DPO:")
print(f"Loss DPO mới: {loss.item():.6f}")
print(f"Biên phần thưởng ẩn E[r(y_w) - r(y_l)]: +{margin.item():.4f}")
print(f"Trọng số đạo hàm biên trung bình sigma(r_l - r_w): {gw.item():.4f}")
print(f"Độ chính xác sở thích (r_w > r_l): {100*acc.item():.1f}%")
