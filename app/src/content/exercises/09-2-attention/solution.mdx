# Các chữ số từ 0 đến 9 là các token của chính chúng. Bộ giải mã
# đọc một token bắt đầu (10) trước khi viết chữ số đầu tiên.

START = 10

# Một lô gồm n chuỗi chữ số ngẫu nhiên có cùng độ dài và các
# "bản dịch" của chúng: chính các chữ số đó theo thứ tự đảo ngược
# [Sao chép từ phần khởi đầu bài 9.1]
import torch

def make_batch(n, length):
    sources = torch.randint(0, 10, (n, length))
    targets = sources.flip(1)   # đảo ngược thứ tự các cột
    return sources, targets

torch.manual_seed(0)
sources, targets = make_batch(3, 8)
for source, target in zip(sources.tolist(), targets.tolist()):
    source_text = "".join(str(d) for d in source)
    target_text = "".join(str(d) for d in target)
    print(f"{source_text}  ->  {target_text}")


# (1) Mô hình. Một mạng GRU mã hóa đọc các chữ số nguồn, và trạng
# thái cuối cùng của nó làm trạng thái đầu tiên của mạng GRU giải mã.
# Bộ giải mã đọc token bắt đầu cùng các chữ số đã viết cho đến hiện
# tại, rồi đưa ra điểm số cho chữ số tiếp theo ở mỗi vị trí.
# [Sao chép từ lời giải bài 9.1, bước (1)]
from torch import nn
import torch.nn.functional as F

class EncoderDecoder(nn.Module):
    def __init__(self, D, H):
        super().__init__()
        self.source_embedding = nn.Embedding(10, D)
        self.encoder = nn.GRU(D, H, batch_first=True)
        self.target_embedding = nn.Embedding(11, D)  # có thêm START
        self.decoder = nn.GRU(D, H, batch_first=True)
        self.output = nn.Linear(H, 10)

    def forward(self, sources, decoder_inputs):
        e = self.source_embedding(sources)          # (batch, L, D)
        _, h = self.encoder(e)       # trạng thái cuối: (1, batch, H)
        d = self.target_embedding(decoder_inputs)   # (batch, N, D)
        states, _ = self.decoder(d, h)              # (batch, N, H)
        return self.output(states)                  # (batch, N, 10)


# (2) Huấn luyện với cưỡng bức giáo viên (teacher forcing): bộ giải mã
# đọc các chữ số đích thực sự, dịch sang phải một vị trí sau token
# bắt đầu. Mỗi lô có độ dài riêng từ 5 đến 30 nên không cần đệm.
# Viết thành hàm vì bài 9.2 sẽ huấn luyện thêm một mô hình nữa.
# [Sao chép từ lời giải bài 9.1, bước (2)]
import random

def train(model, steps=20_000, batch_size=64):
    optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
    total_loss = 0
    for step in range(steps):
        length = random.randint(5, 30)
        sources, targets = make_batch(batch_size, length)
        starts = torch.full((batch_size, 1), START)
        decoder_inputs = torch.cat([starts, targets[:, :-1]], dim=1)
        scores = model(sources, decoder_inputs)
        loss = F.cross_entropy(scores.reshape(-1, 10),
                               targets.reshape(-1))
        optimizer.zero_grad()
        loss.backward()
        nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        optimizer.step()
        total_loss += loss.item()
        if (step + 1) % 5000 == 0:   # trung bình của 5.000 lô
            print(f"batches {step - 4998} to {step + 1}: "
                  f"loss {total_loss / 5000:.3f}")
            total_loss = 0

torch.manual_seed(0)
random.seed(0)
model = EncoderDecoder(D=32, H=128)
train(model)


# (3) Giải mã tham lam: bắt đầu từ token START và viết ra chữ số có
# điểm số cao nhất, từng chữ số một. Ở mỗi bước, mô hình đọc lại
# toàn bộ các chữ số đã viết: hơi lãng phí nhưng đơn giản và đủ
# nhanh cho 30 chữ số.
# [Sao chép từ lời giải bài 9.1, bước (3)]
def translate(model, sources):
    decoder_inputs = torch.full((len(sources), 1), START)
    with torch.no_grad():
        for _ in range(sources.shape[1]):
            scores = model(sources, decoder_inputs)  # (batch, N, 10)
            next_digits = scores[:, -1].argmax(dim=1)
            decoder_inputs = torch.cat(
                [decoder_inputs, next_digits[:, None]], dim=1)
    return decoder_inputs[:, 1:]   # bỏ token bắt đầu


# (4) Với mỗi độ dài, tính tỷ lệ chuỗi hoàn toàn không sai và tỷ lệ
# chữ số đúng trên 1.000 chuỗi mới
# [Sao chép từ lời giải bài 9.1, bước (4)]
lengths = list(range(5, 31))

def accuracy_by_length(model):
    string_accuracy, digit_accuracy = [], []
    for length in lengths:
        sources, targets = make_batch(1000, length)
        correct = translate(model, sources) == targets  # (1000, L)
        whole = correct.all(dim=1)   # một giá trị cho mỗi chuỗi
        string_accuracy.append(whole.float().mean().item())
        digit_accuracy.append(correct.float().mean().item())
    return string_accuracy, digit_accuracy

torch.manual_seed(1)
string_accuracy, digit_accuracy = accuracy_by_length(model)
print()
for length in [10, 15, 20, 25, 30]:

    print(f"{length} digits: "
          f"{string_accuracy[i]:.1%} of the strings, "
          f"{digit_accuracy[i]:.1%} of the digits")

# (1) Mô hình ở bài 9.1 có thêm cơ chế chú ý (attention). Tầng đầu ra
# đọc trạng thái của bộ giải mã cùng một vector ngữ cảnh: trung bình
# của tất cả trạng thái bộ mã hóa, có trọng số là softmax của tích vô
# hướng giữa chúng với trạng thái bộ giải mã. Chỉ thay đổi tầng đầu
# ra và hàm forward().
class AttentionEncoderDecoder(EncoderDecoder):
    def __init__(self, D, H):
        super().__init__(D, H)
        self.output = nn.Linear(2 * H, 10)   # trạng thái và ngữ cảnh

    def forward(self, sources, decoder_inputs):
        e = self.source_embedding(sources)          # (batch, L, D)
        encoder_states, h = self.encoder(e)         # (batch, L, H)
        d = self.target_embedding(decoder_inputs)   # (batch, N, D)
        states, _ = self.decoder(d, h)              # (batch, N, H)
        # Mỗi trạng thái giải mã nhân với mỗi trạng thái mã hóa: (batch, N, L)
        scores = states @ encoder_states.transpose(1, 2)
        weights = torch.softmax(scores, dim=2)  # mỗi hàng có tổng bằng 1
        contexts = weights @ encoder_states         # (batch, N, H)
        self.weights = weights   # lưu lại để vẽ ở bước (4)
        both = torch.cat([states, contexts], dim=2)   # (batch, N, 2H)
        return self.output(both)                    # (batch, N, 10)
