```python
# (1) Hold out 5,000 training reviews as a validation set, and
# replace the HTML line breaks with spaces

def clean(texts):
    return [text.replace("<br />", " ") for text in texts]

split = imdb["train"].train_test_split(test_size=5000, seed=0)
train_texts = clean(split["train"]["text"])
val_texts = clean(split["test"]["text"])
test_texts = clean(imdb["test"]["text"])
train_labels = split["train"]["label"]
val_labels = split["test"]["label"]
test_labels = imdb["test"]["label"]

print(f"train: {len(train_texts)} reviews")
print(f"validation: {len(val_texts)} reviews")
print(f"test: {len(test_texts)} reviews")


# (2) Count the words of each review, over the 10,000 most
# frequent words of the training reviews. The pattern makes every
# run of letters or digits a word, even a single letter like "a".
from sklearn.feature_extraction.text import CountVectorizer

vectorizer = CountVectorizer(max_features=10000,
                             token_pattern=r"\b\w+\b")
X_train = vectorizer.fit_transform(train_texts)
X_val = vectorizer.transform(val_texts)
X_test = vectorizer.transform(test_texts)


# (3) Train a logistic regression for several values of C, the
# inverse of the strength of the regularization, and keep the best
# model on the validation set
from sklearn.linear_model import LogisticRegression

print()
best_accuracy = 0
for C in [0.01, 0.03, 0.1, 0.3, 1, 3, 10]:
    model = LogisticRegression(C=C, max_iter=1000)
    model.fit(X_train, train_labels)
    accuracy = model.score(X_val, val_labels)
    print(f"C = {C}: validation accuracy {accuracy:.3f}")
    if accuracy > best_accuracy:
        best_accuracy, best_C, best_model = accuracy, C, model


# (4) Measure the test accuracy of the best model
test_accuracy = best_model.score(X_test, test_labels)
print(f"\ntest accuracy, C = {best_C}: {test_accuracy:.3f}")


# (5) Print the words with the largest weights, toward negative
# and toward positive. np.argsort sorts the weights in increasing
# order and returns their positions.
import numpy as np

words = vectorizer.get_feature_names_out()
weights = best_model.coef_[0]   # one weight per word
order = np.argsort(weights)
most_negative = [words[i] for i in order[:10]]
most_positive = [words[i] for i in order[-10:]]
most_positive.reverse()
print("\nnegative:", " ".join(most_negative))
print("positive:", " ".join(most_positive))
```
