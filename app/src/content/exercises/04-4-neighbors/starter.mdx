from datasets import load_dataset

imdb = load_dataset("stanfordnlp/imdb")


# (1) Tách 5.000 bài đánh giá huấn luyện làm tập xác thực, và
# thay thế các thẻ xuống dòng HTML bằng khoảng trắng
# [Sao chép từ lời giải bài 3.2, bước (1)]

def clean(texts):
    return [text.replace("<br />", " ") for text in texts]

split = imdb["train"].train_test_split(test_size=5000, seed=0)
train_texts = clean(split["train"]["text"])
val_texts = clean(split["test"]["text"])
test_texts = clean(imdb["test"]["text"])
train_labels = split["train"]["label"]
val_labels = split["test"]["label"]
test_labels = imdb["test"]["label"]

print(f"train: {len(train_texts)} reviews")
print(f"validation: {len(val_texts)} reviews")
print(f"test: {len(test_texts)} reviews")


# (2) Đếm từ trong mỗi bài đánh giá trên 10.000 từ phổ biến nhất
# của tập huấn luyện. Biểu thức chính quy coi mọi chuỗi chữ cái
# hoặc chữ số là một từ, kể cả từ chỉ gồm một chữ cái như "a".
# [Sao chép từ lời giải bài 3.2, bước (2)]
from sklearn.feature_extraction.text import CountVectorizer

vectorizer = CountVectorizer(max_features=10000,
                             token_pattern=r"\b\w+\b")
X_train = vectorizer.fit_transform(train_texts)
X_val = vectorizer.transform(val_texts)
X_test = vectorizer.transform(test_texts)


# (1) Chuyển các vector đếm thành vector tần suất: chia số đếm
# cho tổng số từ của bài đánh giá. X.toarray() tạo mảng numpy từ
# các hàng của ma trận thưa: chỉ xử lý từng nhóm hàng mỗi lần vì
# chuyển toàn bộ cùng lúc sẽ tốn hàng gigabyte bộ nhớ.
# [Sao chép từ lời giải bài 4.3, bước (1)]
import torch

def frequencies(X):
    counts = torch.tensor(X.toarray(), dtype=torch.float32)
    lengths = counts.sum(dim=1, keepdim=True)
    return counts / lengths


# (2) Xây dựng mạng nơ-ron. Tầng đầu tiên (không có hệ số chệch)
# nhân vector tần suất với ma trận W kích thước D x V: kết quả là
# trung bình cộng các vector nhúng của các từ trong bài đánh giá.
# Tiếp theo là hai tầng ẩn và một điểm số đầu ra.
# [Sao chép từ lời giải bài 4.3, bước (2)]
from torch import nn

V = X_train.shape[1]   # kích thước từ vựng
D = 100   # kích thước vector nhúng
H = 100   # kích thước các tầng ẩn
torch.manual_seed(0)
model = nn.Sequential(
    nn.Linear(V, D, bias=False),   # trung bình các vector nhúng
    nn.Linear(D, H),
    nn.ReLU(),
    nn.Linear(H, H),
    nn.ReLU(),
    nn.Linear(H, 1),   # p(tích cực) = sigmoid(điểm số)
)


# (3) Huấn luyện bằng Adam trên từng lô 64 bài đánh giá với hàm
# mất mát entropy chéo cho hai lớp. Sau mỗi epoch, đo độ chính xác
# trên tập xác thực và lưu bản sao của mô hình tốt nhất.
# [Sao chép từ lời giải bài 4.3, bước (3)]
import copy

def predict(model, X):
    # Tính p(tích cực) cho mỗi bài trong X, mỗi lần 1.000 bài
    probabilities = []
    with torch.no_grad():
        for start in range(0, X.shape[0], 1000):
            scores = model(frequencies(X[start:start + 1000]))
            probabilities.append(torch.sigmoid(scores[:, 0]))
    return torch.cat(probabilities)

def accuracy(model, X, labels):
    right = (predict(model, X) > 0.5) == torch.tensor(labels)
    return right.float().mean().item()

y_train = torch.tensor(train_labels, dtype=torch.float32)
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
loss_function = nn.BCEWithLogitsLoss()   # sigmoid + entropy chéo
batch_size = 64
best_accuracy = 0
for epoch in range(5):
    order = torch.randperm(len(y_train)).tolist()   # xáo trộn
    for start in range(0, len(order), batch_size):
        batch = order[start:start + batch_size]
        scores = model(frequencies(X_train[batch]))[:, 0]
        loss = loss_function(scores, y_train[batch])
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
    val_accuracy = accuracy(model, X_val, val_labels)
    print(f"epoch {epoch + 1}: "
          f"validation accuracy {val_accuracy:.3f}")
    if val_accuracy > best_accuracy:
        best_accuracy, best_epoch = val_accuracy, epoch + 1
        best_model = copy.deepcopy(model)
