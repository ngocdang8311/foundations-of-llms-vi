from datasets import load_dataset

imdb = load_dataset("stanfordnlp/imdb")


# (1) Hold out 5,000 training reviews as a validation set, and
# replace the HTML line breaks with spaces
# [Copied from the solution of 3.2, step (1)]

def clean(texts):
    return [text.replace("<br />", " ") for text in texts]

split = imdb["train"].train_test_split(test_size=5000, seed=0)
train_texts = clean(split["train"]["text"])
val_texts = clean(split["test"]["text"])
test_texts = clean(imdb["test"]["text"])
train_labels = split["train"]["label"]
val_labels = split["test"]["label"]
test_labels = imdb["test"]["label"]

print(f"train: {len(train_texts)} reviews")
print(f"validation: {len(val_texts)} reviews")
print(f"test: {len(test_texts)} reviews")


# (2) Count the words of each review, over the 10,000 most
# frequent words of the training reviews. The pattern makes every
# run of letters or digits a word, even a single letter like "a".
# [Copied from the solution of 3.2, step (2)]
from sklearn.feature_extraction.text import CountVectorizer

vectorizer = CountVectorizer(max_features=10000,
                             token_pattern=r"\b\w+\b")
X_train = vectorizer.fit_transform(train_texts)
X_val = vectorizer.transform(val_texts)
X_test = vectorizer.transform(test_texts)


# (1) Turn count vectors into frequency vectors: the counts
# divided by the number of words of the review. X.toarray() makes
# a numpy array from rows of a sparse matrix: do it a few rows at a
# time, since all the rows at once would take gigabytes.
# [Copied from the solution of 4.3, step (1)]
import torch

def frequencies(X):
    counts = torch.tensor(X.toarray(), dtype=torch.float32)
    lengths = counts.sum(dim=1, keepdim=True)
    return counts / lengths


# (2) Build the network. Its first layer, without bias, multiplies
# the frequency vector by the matrix W of size D x V: this gives
# the average of the embeddings of the words of the review. Then
# two hidden layers, and one output score.
# [Copied from the solution of 4.3, step (2)]
from torch import nn

V = X_train.shape[1]   # size of the vocabulary
D = 100   # size of the embeddings
H = 100   # size of the hidden layers
torch.manual_seed(0)
model = nn.Sequential(
    nn.Linear(V, D, bias=False),   # average of the embeddings
    nn.Linear(D, H),
    nn.ReLU(),
    nn.Linear(H, H),
    nn.ReLU(),
    nn.Linear(H, 1),   # p(positive) = sigmoid(score)
)


# (3) Train with Adam, on batches of 64 reviews, with the
# cross-entropy loss for two classes. After each epoch, measure
# the validation accuracy, and keep a copy of the best model.
# [Copied from the solution of 4.3, step (3)]
import copy

def predict(model, X):
    # p(positive) for each review of X, 1,000 reviews at a time
    probabilities = []
    with torch.no_grad():
        for start in range(0, X.shape[0], 1000):
            scores = model(frequencies(X[start:start + 1000]))
            probabilities.append(torch.sigmoid(scores[:, 0]))
    return torch.cat(probabilities)

def accuracy(model, X, labels):
    right = (predict(model, X) > 0.5) == torch.tensor(labels)
    return right.float().mean().item()

y_train = torch.tensor(train_labels, dtype=torch.float32)
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
loss_function = nn.BCEWithLogitsLoss()   # sigmoid + cross-entropy
batch_size = 64
best_accuracy = 0
for epoch in range(5):
    order = torch.randperm(len(y_train)).tolist()   # shuffled
    for start in range(0, len(order), batch_size):
        batch = order[start:start + batch_size]
        scores = model(frequencies(X_train[batch]))[:, 0]
        loss = loss_function(scores, y_train[batch])
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
    val_accuracy = accuracy(model, X_val, val_labels)
    print(f"epoch {epoch + 1}: "
          f"validation accuracy {val_accuracy:.3f}")
    if val_accuracy > best_accuracy:
        best_accuracy, best_epoch = val_accuracy, epoch + 1
        best_model = copy.deepcopy(model)
