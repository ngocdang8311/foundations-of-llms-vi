```python
# (1) Cài đặt hàm tính điểm BM25 trên kho tài liệu
N = len(DOCS)
doc_tokens = [d["text"].split() for d in DOCS]
avgdl = sum(len(toks) for toks in doc_tokens) / N

def bm25_rank(query_str, k1=1.5, b=0.75):
    q_terms = query_str.split()
    scores = []
    for i, toks in enumerate(doc_tokens):
        dl = len(toks)
        s = 0.0
        for w in q_terms:
            df = sum(1 for dt in doc_tokens if w in dt)
            idf = math.log(1.0 + (N - df + 0.5) / (df + 0.5))
            tf = toks.count(w)
            num = tf * (k1 + 1.0)
            den = tf + k1 * (1.0 - b + b * (dl / avgdl))
            s += idf * (num / den)
        scores.append((DOCS[i]["id"], s))
    return [doc_id for doc_id, _ in sorted(scores, key=lambda x: -x[1])]


# (2) Xếp hạng theo tương đồng Cosine và hợp nhất bằng RRF
def dense_rank(q_vec):
    q = np.array(q_vec)
    scores = [(d["id"], float(np.dot(q, d["vec"]))) for d in DOCS]
    return [doc_id for doc_id, _ in sorted(scores, key=lambda x: -x[1])]

def rrf_fusion(rank_a, rank_b, k=2):
    rrf = {}
    for r, doc_id in enumerate(rank_a, start=1):
        rrf[doc_id] = rrf.get(doc_id, 0.0) + 1.0 / (k + r)
    for r, doc_id in enumerate(rank_b, start=1):
        rrf[doc_id] = rrf.get(doc_id, 0.0) + 1.0 / (k + r)
    return [doc_id for doc_id, _ in sorted(rrf.items(), key=lambda x: -x[1])]


# (3) Đánh giá Recall@2 trên tập câu hỏi
recalls = {"bm25": [], "dense": [], "rrf": []}
for i, item in enumerate(QUERIES, start=1):
    r_bm = bm25_rank(item["q"])
    r_de = dense_rank(item["q_vec"])
    r_hy = rrf_fusion(r_bm, r_de, k=2)
    print(f"Query {i} top-2 -> BM25: {r_bm[:2]} | Dense: {r_de[:2]} | RRF: {r_hy[:2]}")
    for name, ranked in [("bm25", r_bm), ("dense", r_de), ("rrf", r_hy)]:
        hit = len(set(ranked[:2]) & item["gold"]) / len(item["gold"])
        recalls[name].append(hit)

print(
    f"Recall@2      -> BM25: {np.mean(recalls['bm25'])*100:.1f}% | "
    f"Dense: {np.mean(recalls['dense'])*100:.1f}% | "
    f"Hybrid RRF: {np.mean(recalls['rrf'])*100:.1f}%"
)
```
