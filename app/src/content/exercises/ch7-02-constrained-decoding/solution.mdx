# (1) Kiểm tra xem việc nối thêm token ứng viên có thỏa văn phạm không
def is_valid_transition(prefix, candidate_tok):
    new_text = prefix + candidate_tok
    return any(re.match(pat, new_text) for pat in VALID_PREFIX_PATTERNS)


# (2) Áp mặt nạ logit -inf cho các token làm hỏng cấu trúc JSON
def grammar_masked_logits(prefix, logits):
    masked = logits.copy()
    for idx, tok in enumerate(VOCAB):
        if not is_valid_transition(prefix, tok):
            masked[idx] = -np.inf
    return masked


# (3) So sánh giải mã tham lam không ràng buộc và có ràng buộc FSM
unconstrained = "".join(VOCAB[int(np.argmax(row))] for row in RAW_LOGITS)
try:
    json.loads(unconstrained)
    unconstrained_ok = True
except Exception:
    unconstrained_ok = False

prefix = ""
for row in RAW_LOGITS:
    masked = grammar_masked_logits(prefix, row)
    best_id = int(np.argmax(masked))
    prefix += VOCAB[best_id]

parsed = json.loads(prefix)
print(f"Unconstrained output : {unconstrained}")
print(f"Unconstrained valid? : {unconstrained_ok}")
print(f"Constrained output   : {prefix}")
print(
    f"Constrained valid?   : True -> parsed tool='{parsed['tool']}', "
    f"expr='{parsed['expr']}'"
)
