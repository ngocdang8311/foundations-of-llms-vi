```python
from datasets import load_dataset

imdb = load_dataset("stanfordnlp/imdb")


# (1) Giữ lại 5.000 bài đánh giá huấn luyện làm tập xác thực và
# thay thế các thẻ ngắt dòng HTML bằng khoảng trắng

def clean(texts):
    return [text.replace("<br />", " ") for text in texts]

split = imdb["train"].train_test_split(test_size=5000, seed=0)
train_texts = clean(split["train"]["text"])
val_texts = clean(split["test"]["text"])

train_labels = split["train"]["label"]
val_labels = split["test"]["label"]
```

```python
# (1) Ma trận nhúng tiền huấn luyện: cột j là vector GloVe của từ j
# trong từ vựng, theo thứ tự của các vector đếm. Một vài từ mà
# GloVe không biết sẽ giữ nguyên cột toàn số 0.
import torch

glove_index = {word: i for i, word in enumerate(glove_words)}
vocabulary = vectorizer.get_feature_names_out()
D = glove_vectors.shape[1]   # 300
W_glove = torch.zeros(D, len(vocabulary))
missing = []
for j, word in enumerate(vocabulary):
    if word in glove_index:
        W_glove[:, j] = glove_vectors[glove_index[word]]
    else:
        missing.append(word)
print(f"{len(missing)} words without a GloVe vector: {missing[:5]}")


# (2) Mạng trung bình sâu ở bài 4.3, với D = 300. Tầng đầu tiên là W:
# ngẫu nhiên (từ đầu), hoặc dùng vector GloVe giữ cố định (đóng băng)
# hay huấn luyện cùng phần còn lại (tinh chỉnh).
from torch import nn

def frequencies(X):
    counts = torch.tensor(X.toarray(), dtype=torch.float32)
    lengths = counts.sum(dim=1, keepdim=True)
    return counts / lengths

def make_model(embeddings):
    torch.manual_seed(0)
    model = nn.Sequential(
        nn.Linear(len(vocabulary), D, bias=False),   # trung bình
        nn.Linear(D, 100),
        nn.ReLU(),
        nn.Linear(100, 100),
        nn.ReLU(),
        nn.Linear(100, 1),   # p(positive) = sigmoid(score)
    )
    if embeddings != "from scratch":
        model[0].weight.data = W_glove.clone()
    if embeddings == "frozen":
        model[0].weight.requires_grad_(False)   # không tính đạo hàm
    return model


# (3) Hàm huấn luyện mô hình qua 1.500 lô gồm 64 bài đánh giá lấy
# ngẫu nhiên từ tập huấn luyện. Sau mỗi 100 lô, đo độ chính xác xác
# thực và giữ lại bản sao của mô hình tốt nhất.
import copy

def predict(model, X):
    # p(positive) cho mỗi bài đánh giá trong X, mỗi lần 1.000 bài
    probabilities = []
    with torch.no_grad():
        for start in range(0, X.shape[0], 1000):
            scores = model(frequencies(X[start:start + 1000]))
            probabilities.append(torch.sigmoid(scores[:, 0]))
    return torch.cat(probabilities)

def accuracy(model, X, labels):
    right = (predict(model, X) > 0.5) == torch.tensor(labels)
    return right.float().mean().item()

def train(model, X, labels, n_batches=1500):
    y = torch.tensor(labels, dtype=torch.float32)
    trained = [p for p in model.parameters() if p.requires_grad]
    optimizer = torch.optim.Adam(trained, lr=1e-3)
    loss_function = nn.BCEWithLogitsLoss()
    best_accuracy = 0
    for step in range(1, n_batches + 1):
        batch = torch.randint(len(y), (64,)).tolist()
        scores = model(frequencies(X[batch]))[:, 0]
        loss = loss_function(scores, y[batch])
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
        if step % 100 == 0:
            val_accuracy = accuracy(model, X_val, val_labels)
            if val_accuracy > best_accuracy:
                best_accuracy = val_accuracy
                best_model = copy.deepcopy(model)
    return best_model


# (4) Huấn luyện ba phiên bản và đo độ chính xác trên tập kiểm tra
models = {}
for embeddings in ["from scratch", "frozen", "fine-tuned"]:
    models[embeddings] = train(make_model(embeddings),
                               X_train, train_labels)
    test_accuracy = accuracy(models[embeddings], X_test, test_labels)
    print(f"{embeddings:12}  test accuracy {test_accuracy:.3f}")


# (5) Các láng giềng gần nhất của "good" và "bad" trong từ vựng,
# trên GloVe và trên W sau khi tinh chỉnh
def neighbors(W, word, n=5):
    W = W / W.norm(dim=0, keepdim=True).clamp(min=1e-8)
    similarities = W[:, vectorizer.vocabulary_[word]] @ W
    ranking = similarities.argsort(descending=True).tolist()
    return [vocabulary[j] for j in ranking[1:n + 1]]   # [0]: chính từ đó

W_tuned = models["fine-tuned"][0].weight.detach()
print()
for word in ["good", "bad"]:
    print(f"{word}, GloVe:      {', '.join(neighbors(W_glove, word))}")
    print(f"{word}, fine-tuned: {', '.join(neighbors(W_tuned, word))}")
```
