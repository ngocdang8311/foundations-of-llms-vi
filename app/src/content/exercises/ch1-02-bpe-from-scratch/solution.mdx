# (1) Hàm đếm tần suất cặp kề nhau và hàm gộp cặp thành ID mới
def get_stats(ids):
    counts = {}
    for pair in zip(ids, ids[1:]):
        counts[pair] = counts.get(pair, 0) + 1
    return counts


def merge_vocab(ids, pair, new_id):
    new_ids, i = [], 0
    while i < len(ids):
        if i < len(ids) - 1 and (ids[i], ids[i + 1]) == pair:
            new_ids.append(new_id)
            i += 2
        else:
            new_ids.append(ids[i])
            i += 1
    return new_ids


# (2) Huấn luyện BPE trong 200 bước gộp từ bảng mã 256 byte cơ sở
ids = list(raw_bytes)
merges = {}
vocab = {i: bytes([i]) for i in range(256)}

for step in range(200):
    stats = get_stats(ids)
    best_pair = max(stats, key=stats.get)
    new_id = 256 + step
    ids = merge_vocab(ids, best_pair, new_id)
    merges[best_pair] = new_id
    vocab[new_id] = vocab[best_pair[0]] + vocab[best_pair[1]]
    if step in (0, 1, 2, 9, 49, 99, 149, 199):
        s = vocab[new_id].decode("utf-8", errors="replace")
        freq = stats[best_pair]
        print(f"Bước {step+1:3d}: gộp {best_pair} -> {new_id} "
              f"({s!r}, tần suất {freq}), còn {len(ids)} tokens")

ratio = len(raw_bytes) / len(ids)
print(f"Tỷ lệ nén sau 200 bước: "
      f"{len(raw_bytes)}/{len(ids)} = {ratio:.2f} bytes/token")


# (3) Hàm mã hóa (encode) theo thứ tự ưu tiên gộp và giải mã (decode)
def encode(text):
    tokens = list(text.encode("utf-8"))
    while len(tokens) >= 2:
        stats = get_stats(tokens)
        pair = min(stats, key=lambda p: merges.get(p, float("inf")))
        if pair not in merges:
            break
        tokens = merge_vocab(tokens, pair, merges[pair])
    return tokens


def decode(token_ids):
    raw = b"".join(vocab[idx] for idx in token_ids)
    return raw.decode("utf-8", errors="replace")


sample = "Simple language models read subword tokens."
