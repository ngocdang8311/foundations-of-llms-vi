from datasets import load_dataset

wikitext = load_dataset("Salesforce/wikitext", "wikitext-2-v1")


# (1) Tạo chuỗi huấn luyện bằng cách tách từ ở mỗi dòng theo dấu cách
# [Sao chép từ lời giải bài 2.1, bước (1)]
train_sequence = []
for line in wikitext["train"]["text"]:
    train_sequence.extend(line.split())

print(f"text: {len(train_sequence)} words")
print(f"vocabulary: {len(set(train_sequence))} words")


# Chuỗi xác thực (validation) và kiểm tra (test), tạo tương tự
# [Sao chép từ mã khởi đầu bài 2.2]
val_sequence = []
for line in wikitext["validation"]["text"]:
    val_sequence.extend(line.split())

test_sequence = []
for line in wikitext["test"]["text"]:
    test_sequence.extend(line.split())

print(f"validation: {len(val_sequence)} words")
print(f"test: {len(test_sequence)} words")


# (1) Với mỗi n, đếm các n-gram trong một từ điển lồng nhau:
# counts[n][context][word] là số lần `word` xuất hiện ngay sau
# `context` (tuple gồm n - 1 từ đứng trước nó).
# Với n = 1, ngữ cảnh là tuple rỗng ().
# [Sao chép từ lời giải bài 2.3a, bước (1)]
counts = {}
for n in range(1, 6):
    counts[n] = {}
    for i in range(n - 1, len(train_sequence)):
        context = tuple(train_sequence[i - n + 1:i])
        word = train_sequence[i]
        if context not in counts[n]:
            counts[n][context] = {}
        if word not in counts[n][context]:
            counts[n][context][word] = 0
        counts[n][context][word] += 1

# (1) Với mỗi từ trong chuỗi và mỗi n, tra cứu hai số đếm từ tập
# huấn luyện: số lần xuất hiện của n-gram kết thúc tại từ đó, và
# số lần xuất hiện của ngữ cảnh. Bắt đầu từ từ thứ 5 để mọi từ
# đều có đủ 4 từ đứng trước. Lưu kết quả vào mảng numpy.
import numpy as np

context_totals = {}
for n in range(1, 6):
    context_totals[n] = {}
    for context in counts[n]:
        context_totals[n][context] = sum(counts[n][context].values())

def lookup_counts(sequence):
    ngram_count = {}
    context_count = {}
    for n in range(1, 6):
        ngram_list = []
        context_list = []
        for i in range(4, len(sequence)):
            context = tuple(sequence[i - n + 1:i])
            followers = counts[n].get(context, {})
            ngram_list.append(followers.get(sequence[i], 0))
            context_list.append(context_totals[n].get(context, 0))
        ngram_count[n] = np.array(ngram_list)
        context_count[n] = np.array(context_list)
    return ngram_count, context_count

val_ngram, val_context = lookup_counts(val_sequence)

def perplexity(probs):
    return np.exp(-np.mean(np.log(probs)))


# (2) Làm mịn cộng một (add-one): cộng 1 vào số đếm của mọi
# n-gram khả dĩ, nên tổng số đếm của mỗi ngữ cảnh tăng thêm V
# (kích thước từ vựng)
V = len(set(train_sequence))
add_one = []
for n in range(1, 6):
    probs = (val_ngram[n] + 1) / (val_context[n] + V)
    add_one.append(perplexity(probs))
    print(f"n = {n}: add-one perplexity {add_one[-1]:.1f}")


# (3) Nội suy: q_n = w * p_n + (1 - w) * q_(n-1), trong đó p_n
# là xác suất tính từ số đếm n-gram. Khi ngữ cảnh chưa từng gặp
# trong tập huấn luyện, đặt q_n = q_(n-1).
def interpolate(ngram_count, context_count, weights):
    q = ngram_count[1] / context_count[1]   # n = 1
    for n in range(2, len(weights) + 2):
        w = weights[n - 2]   # weights[0] là trọng số của n = 2
        seen = context_count[n] > 0
        p = np.zeros(len(q))
        p[seen] = ngram_count[n][seen] / context_count[n][seen]
        q = np.where(seen, w * p + (1 - w) * q, q)
    return q

# Tinh chỉnh lần lượt trọng số của từng n bằng tìm kiếm lưới
weights = []
interpolation = [perplexity(interpolate(val_ngram, val_context, []))]
print()
for n in range(2, 6):
    best_w, best_perplexity = 0, np.inf
    for k in range(1, 100):
        w = k / 100
        q = interpolate(val_ngram, val_context, weights + [w])
        if perplexity(q) < best_perplexity:
            best_w, best_perplexity = w, perplexity(q)
    weights.append(best_w)
    interpolation.append(best_perplexity)
    print(f"n = {n}: weight {best_w}, "
          f"interpolated perplexity {best_perplexity:.1f}")


# (4) Độ bối rối trên tập kiểm tra của mô hình 5-gram nội suy
test_ngram, test_context = lookup_counts(test_sequence)
q = interpolate(test_ngram, test_context, weights)
test_perplexity = perplexity(q)
print(f"\ntest perplexity, n = 5: {test_perplexity:.1f}")
