```python
from datasets import load_dataset

wikitext = load_dataset("Salesforce/wikitext", "wikitext-2-v1")


# (1) Tạo chuỗi huấn luyện bằng cách tách từ ở mỗi dòng theo dấu cách
# [Sao chép từ lời giải bài 2.1, bước (1)]
train_sequence = []
for line in wikitext["train"]["text"]:
    train_sequence.extend(line.split())

print(f"text: {len(train_sequence)} words")
print(f"vocabulary: {len(set(train_sequence))} words")


# (2) Dùng từ điển lồng nhau để đếm các cặp từ (bigram)
# [Sao chép từ lời giải bài 2.1, bước (2)]
bigram_counts = {}
for a, b in zip(train_sequence, train_sequence[1:]):
    if a not in bigram_counts:
        bigram_counts[a] = {}
    if b not in bigram_counts[a]:
        bigram_counts[a][b] = 0
    bigram_counts[a][b] += 1

print(f"distinct pairs: {sum(len(f) for f in bigram_counts.values())}")


# -------------------------
#  Phần mới trong bài này
# -------------------------
# Chuỗi xác thực (validation) và kiểm tra (test), tạo tương tự
val_sequence = []
for line in wikitext["validation"]["text"]:
    val_sequence.extend(line.split())

test_sequence = []
for line in wikitext["test"]["text"]:
    test_sequence.extend(line.split())

print(f"validation: {len(val_sequence)} words")
print(f"test: {len(test_sequence)} words")
```

```python
# (1) Đếm số lần xuất hiện của từng từ trong chuỗi huấn luyện
unigram_counts = {}
for word in train_sequence:
    if word not in unigram_counts:
        unigram_counts[word] = 0
    unigram_counts[word] += 1


# (2) Độ bối rối (perplexity) của mô hình unigram: hàm mũ của trừ
# trung bình log-xác suất trên các từ thuộc tập xác thực
import math

def perplexity(log_probs):
    return math.exp(-sum(log_probs) / len(log_probs))

n_train = len(train_sequence)
log_probs = []
for word in val_sequence:
    log_probs.append(math.log(unigram_counts[word] / n_train))

print(f"unigram perplexity, validation set: "
      f"{perplexity(log_probs):.1f}")


# (3) Độ bối rối của mô hình bigram. Một cặp từ chưa từng gặp khi
# huấn luyện sẽ có xác suất bằng 0, và log(0) bằng âm vô cùng.
follower_totals = {}
for a in bigram_counts:
    follower_totals[a] = sum(bigram_counts[a].values())

log_probs = []
unseen_pairs = 0
for a, b in zip(val_sequence, val_sequence[1:]):
    count = bigram_counts[a].get(b, 0)
    if count == 0:
        unseen_pairs += 1
        log_probs.append(-math.inf)
    else:
        log_probs.append(math.log(count / follower_totals[a]))

n_pairs = len(log_probs)
print(f"\nvalidation pairs never seen in training: {unseen_pairs}")
print(f"out of {n_pairs}, or {100 * unseen_pairs / n_pairs:.1f}%")
print(f"bigram perplexity, validation set: {perplexity(log_probs)}")


# (4) Độ bối rối của mô hình unigram trên tập kiểm tra (test)
log_probs = []
for word in test_sequence:
    log_probs.append(math.log(unigram_counts[word] / n_train))

test_perplexity = perplexity(log_probs)
print(f"\nunigram perplexity, test set: {test_perplexity:.1f}")
```
