from datasets import load_dataset

imdb = load_dataset("stanfordnlp/imdb")


# (1) Giữ lại 5.000 bài đánh giá huấn luyện làm tập xác thực và

def clean(texts):
    return [text.replace("<br />", " ") for text in texts]

split = imdb["train"].train_test_split(test_size=5000, seed=0)
train_texts = clean(split["train"]["text"])
val_texts = clean(split["test"]["text"])
test_texts = clean(imdb["test"]["text"])
train_labels = split["train"]["label"]
val_labels = split["test"]["label"]

print(f"train: {len(train_texts)} reviews")

# (1) Bộ phân loại: BERT cùng một đầu phân loại đọc vector mới của
# token [CLS] ở vị trí 0. Bộ tách từ đệm các bài đánh giá trong lô
# về cùng độ dài và giữ lại 256 token đầu tiên của các bài dài. Mặt
# nạ chú ý (attention mask) báo cho BERT bỏ qua phần đệm.
import torch
from torch import nn
from transformers import AutoModel

device = "cuda" if torch.cuda.is_available() else "cpu"

class BERTClassifier(nn.Module):
    def __init__(self):
        super().__init__()
        self.bert = AutoModel.from_pretrained("bert-base-uncased")
        self.head = nn.Linear(768, 1)   # p(positive) = sigmoid

    def forward(self, texts):
        batch = tokenizer(texts, truncation=True, max_length=256,
                          padding=True, return_tensors="pt")
        states = self.bert(**batch.to(device)).last_hidden_state
        return self.head(states[:, 0])[:, 0]   # (batch,)

torch.manual_seed(0)
model = BERTClassifier().to(device)


# (2) Tinh chỉnh toàn bộ mạng trên 1.000 bài đánh giá trong 3 epoch,
# theo lô 16 bài với tốc độ học nhỏ. train() bật dropout của BERT,
# còn eval() tắt nó đi.
y_train = torch.tensor(train_labels, dtype=torch.float32)
optimizer = torch.optim.AdamW(model.parameters(), lr=2e-5)
loss_function = nn.BCEWithLogitsLoss()   # sigmoid + cross-entropy
model.train()
random.seed(0)
for epoch in range(3):
    order = list(chosen)
    random.shuffle(order)
    for start in range(0, len(order), 16):
        batch = order[start:start + 16]
        scores = model([train_texts[i] for i in batch])
        loss = loss_function(scores, y_train[batch].to(device))
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
model.eval()


# (3) Độ chính xác trên tập kiểm tra, tính mỗi lần 64 bài đánh giá
def predict(texts):
    probabilities = []
    with torch.no_grad():
        for start in range(0, len(texts), 64):
            scores = model(texts[start:start + 64])
            probabilities.append(torch.sigmoid(scores).cpu())
    return torch.cat(probabilities)

correct = (predict(test_texts) > 0.5) == torch.tensor(test_labels)
test_accuracy = correct.float().mean().item()
print(f"test accuracy: {test_accuracy:.3f}")


# (4) Các bài đánh giá khó, tương tự bài 3.4
p_positive = predict([review for review, label in hard_reviews])

names = ["negative", "positive"]
print("\nlabel     p(positive)  right  review")
right_count = 0
for (review, label), p in zip(hard_reviews, p_positive.tolist()):
    right = (p > 0.5) == (label == 1)
    if right:
        right_count += 1
    mark = "yes" if right else "NO"
    short = review if len(review) <= 36 else review[:35] + "…"
    print(f"{names[label]:9} {p:.3f}        {mark:5}  {short}")
print(f"\n{right_count} right out of {len(hard_reviews)}")
