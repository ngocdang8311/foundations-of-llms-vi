```python
from datasets import load_dataset

imdb = load_dataset("stanfordnlp/imdb")


# (1) Tách 5.000 bài đánh giá huấn luyện làm tập xác thực, và
# thay thế các thẻ xuống dòng HTML bằng khoảng trắng
# [Sao chép từ lời giải bài 3.2, bước (1)]

def clean(texts):
    return [text.replace("<br />", " ") for text in texts]

split = imdb["train"].train_test_split(test_size=5000, seed=0)
train_texts = clean(split["train"]["text"])
val_texts = clean(split["test"]["text"])
test_texts = clean(imdb["test"]["text"])
train_labels = split["train"]["label"]
val_labels = split["test"]["label"]
test_labels = imdb["test"]["label"]

print(f"train: {len(train_texts)} reviews")
print(f"validation: {len(val_texts)} reviews")
print(f"test: {len(test_texts)} reviews")
```

```python
# (1) Ba cách gán trọng số trên 10.000 từ phổ biến nhất
# của các bài đánh giá huấn luyện
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.feature_extraction.text import TfidfVectorizer

V = 10000
pattern = r"\b\w+\b"   # mọi từ, kể cả từ chỉ gồm một chữ cái
vectorizers = {
    "counts": CountVectorizer(max_features=V, token_pattern=pattern),
    "presence": CountVectorizer(max_features=V, token_pattern=pattern,
                                binary=True),
    "TF-IDF": TfidfVectorizer(max_features=V, token_pattern=pattern),
}


# (2) Với mỗi cách gán trọng số, chọn C trên tập xác thực như ở
# bài 3.2, rồi đo độ chính xác trên tập kiểm tra của mô hình tốt nhất
from sklearn.linear_model import LogisticRegression

test_accuracies = {}
for name in vectorizers:
    vectorizer = vectorizers[name]
    X_train = vectorizer.fit_transform(train_texts)
    X_val = vectorizer.transform(val_texts)
    X_test = vectorizer.transform(test_texts)

    best_accuracy = 0
    for C in [0.01, 0.03, 0.1, 0.3, 1, 3, 10]:
        model = LogisticRegression(C=C, max_iter=1000)
        model.fit(X_train, train_labels)
        accuracy = model.score(X_val, val_labels)
        if accuracy > best_accuracy:
            best_accuracy, best_C, best_model = accuracy, C, model

    test_accuracies[name] = best_model.score(X_test, test_labels)
    print(f"{name}: C = {best_C}, "
          f"test accuracy {test_accuracies[name]:.3f}")
```
