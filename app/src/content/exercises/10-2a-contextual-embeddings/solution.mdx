```python
from transformers import AutoTokenizer, AutoModel

# BERT base, huấn luyện trên văn bản chữ thường: 12 tầng, vector 768 chiều
tokenizer = AutoTokenizer.from_pretrained("bert-base-uncased")
model = AutoModel.from_pretrained("bert-base-uncased")

# Mười câu chứa từ "bank": 5 câu đầu mang nghĩa bờ sông, 5 câu sau
# mang nghĩa ngân hàng giữ tiền. Ở một số câu, chỉ những từ đứng sau
# "bank" mới cho biết nghĩa của nó.
sentences = [
    "We had a picnic on the bank of the river.",
    "The boat drifted slowly toward the muddy bank.",
    "Fishermen sat along the bank, waiting for a bite.",
    "After the flood, the river overflowed its bank.",
    "The bank was steep and covered with reeds.",
    "I opened an account at the bank.",
    "The bank raised its interest rates again.",
    "She went to the bank to deposit her salary.",
    "The robbers broke into the bank at night.",
    "The bank refused to give him a loan.",
]
meanings = ["river"] * 5 + ["money"] * 5

# Bộ tách từ cắt câu thành các token, đồng thời thêm [CLS] vào đầu
# câu và [SEP] vào cuối câu
encoding = tokenizer(sentences[0])
print(tokenizer.convert_ids_to_tokens(encoding["input_ids"]))
```

```python
# (1) Vector của từ "bank" trong mỗi câu: tại đầu vào của BERT
# (hidden_states[0], gồm vector nhúng của token cộng với vector
# nhúng vị trí của nó) và sau tầng cuối cùng (hidden_states[12])
import torch

input_vectors, output_vectors = [], []
for sentence in sentences:
    encoding = tokenizer(sentence, return_tensors="pt")
    tokens = tokenizer.convert_ids_to_tokens(encoding["input_ids"][0])
    position = tokens.index("bank")
    with torch.no_grad():
        outputs = model(**encoding, output_hidden_states=True)
    input_vectors.append(outputs.hidden_states[0][0, position])
    output_vectors.append(outputs.hidden_states[12][0, position])


# (2) Độ tương đồng cosine giữa mười vector, vẽ dưới dạng ma trận:
# mỗi câu ứng với một hàng và một cột
import matplotlib.pyplot as plt

def cosine_similarities(vectors):
    V = torch.stack(vectors)
    V = V / V.norm(dim=1, keepdim=True)
    return V @ V.T   # (10, 10)

similarities = {
    "input of BERT": cosine_similarities(input_vectors),
    "last layer": cosine_similarities(output_vectors),
}
fig, axes = plt.subplots(1, 2, figsize=(8, 3.8), layout="constrained")
for ax, (name, S) in zip(axes, similarities.items()):
    image = ax.imshow(S, cmap="Blues", vmin=0, vmax=1)
    ax.set_title(name)
    ax.set_xticks(range(10), meanings, rotation=90)
    ax.set_yticks(range(10), meanings)
fig.colorbar(image, ax=axes, label="cosine similarity")
plt.show()


# (3) Độ tương đồng trung bình giữa hai câu khác nhau có cùng nghĩa
# và có nghĩa khác nhau
for name, S in similarities.items():
    same, different = [], []
    for i in range(10):
        for j in range(i + 1, 10):
            if meanings[i] == meanings[j]:
                same.append(S[i, j].item())
            else:
                different.append(S[i, j].item())
    print(f"{name}: same meaning {sum(same) / len(same):.2f}, "
          f"different meanings {sum(different) / len(different):.2f}")
```
