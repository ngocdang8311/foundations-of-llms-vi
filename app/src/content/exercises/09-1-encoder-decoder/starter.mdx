# The digits 0 to 9 are their own tokens. The decoder reads a start
# token, 10, before it writes the first digit.
START = 10

# A batch of n random strings of digits of the same length, and
# their "translations": the same digits in reverse order
import torch

def make_batch(n, length):
    sources = torch.randint(0, 10, (n, length))
    targets = sources.flip(1)   # reverse the order of the columns
    return sources, targets

torch.manual_seed(0)
sources, targets = make_batch(3, 8)
for source, target in zip(sources.tolist(), targets.tolist()):
    source_text = "".join(str(d) for d in source)
    target_text = "".join(str(d) for d in target)
    print(f"{source_text}  ->  {target_text}")
