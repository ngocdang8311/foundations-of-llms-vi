```python
# (1) Count the words, then subsample the frequent ones: keep each
# occurrence of a word of frequency f with probability sqrt(t / f),
# so that "the" loses most of its occurrences, and a rare word none
import math
import random

counts = {}
for word in train_sequence:
    if word not in counts:
        counts[word] = 0
    counts[word] += 1

t = 1e-4
random.seed(0)
kept = []   # the indices of the words that are kept, in order
for word in train_sequence:
    frequency = counts[word] / len(train_sequence)
    if random.random() < math.sqrt(t / frequency):
        kept.append(word_index[word])
print(f"after subsampling: {len(kept)} words")


# (2) Build the (center, context) pairs: each word of the subsampled
# sequence with each of the 2 words before it and after it.
# sequence[:-offset] and sequence[offset:] pair each word with the
# word that comes offset positions later.
import torch

window = 2
sequence = torch.tensor(kept)
centers, contexts = [], []
for offset in range(1, window + 1):
    centers += [sequence[:-offset], sequence[offset:]]
    contexts += [sequence[offset:], sequence[:-offset]]
centers = torch.cat(centers)
contexts = torch.cat(contexts)
print(f"pairs: {len(centers)}")


# (3) The model: two embedding matrices, one for the words at the
# center (the embeddings we keep), one for the words of the context.
# The score of a pair is the dot product of their two vectors.
from torch import nn

V = len(vocabulary)
D = 100   # size of the embeddings
torch.manual_seed(0)
center_embeddings = nn.Embedding(V, D)
context_embeddings = nn.Embedding(V, D)
nn.init.normal_(center_embeddings.weight, std=0.1)
nn.init.normal_(context_embeddings.weight, std=0.1)


# (4) Train with Adam on batches of 4,096 real pairs. For each real
# pair, draw k = 5 fake context words from the counts raised to the
# power 3/4, with torch.multinomial. The loss is the cross-entropy
# loss for two classes: 1 for the real pairs, 0 for the fake ones.
import time

noise = torch.tensor([counts[word] for word in vocabulary],
                     dtype=torch.float32) ** 0.75
k = 5
parameters = (list(center_embeddings.parameters())
              + list(context_embeddings.parameters()))
optimizer = torch.optim.Adam(parameters, lr=3e-3)
loss_function = nn.BCEWithLogitsLoss()   # sigmoid + cross-entropy
batch_size = 4096
start_time = time.time()
for epoch in range(3):
    order = torch.randperm(len(centers))   # shuffled pairs
    for start in range(0, len(order), batch_size):
        batch = order[start:start + batch_size]
        B = len(batch)
        center = center_embeddings(centers[batch])     # (B, D)
        context = context_embeddings(contexts[batch])  # (B, D)
        fake_words = torch.multinomial(noise, B * k, replacement=True)
        fake = context_embeddings(fake_words).view(B, k, D)
        real_scores = (center * context).sum(dim=1)    # (B,)
        # center[:, None] has size (B, 1, D): it multiplies each of
        # the k fake contexts of its pair
        fake_scores = (center[:, None] * fake).sum(dim=2)  # (B, k)
        scores = torch.cat([real_scores[:, None], fake_scores], dim=1)
        labels = torch.zeros(B, 1 + k)
        labels[:, 0] = 1   # the real pair is in column 0
        loss = loss_function(scores, labels)
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
    elapsed = time.time() - start_time
    print(f"epoch {epoch + 1}: loss {loss.item():.3f}, "
          f"{elapsed:.0f} s since the start")


# (5) The nearest neighbors of the words of exercise 4.2, in the
# matrix of the center words. Divide each row by its length, so
# that a dot product is a cosine similarity.
embeddings = center_embeddings.weight.detach()
embeddings = embeddings / embeddings.norm(dim=1, keepdim=True)

print()
for word in ["three", "he", "city", "film", "excellent", "terrible"]:
    similarities = embeddings @ embeddings[word_index[word]]
    ranking = similarities.argsort(descending=True).tolist()
    nearest = [vocabulary[i] for i in ranking[1:6]]   # [0]: the word
    print(f"{word}: {', '.join(nearest)}")
```
