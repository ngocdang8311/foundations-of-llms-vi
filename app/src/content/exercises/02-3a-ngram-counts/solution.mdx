# (1) For each n, count the n-grams in a nested dict:
# counts[n][context][word] is the number of times `word` follows
# `context`, the tuple of the n - 1 words before it.
# For n = 1, the context is the empty tuple ().
counts = {}
for n in range(1, 6):
    counts[n] = {}
    for i in range(n - 1, len(train_sequence)):
        context = tuple(train_sequence[i - n + 1:i])
        word = train_sequence[i]
        if context not in counts[n]:
            counts[n][context] = {}
        if word not in counts[n][context]:
            counts[n][context][word] = 0
        counts[n][context][word] += 1


# (2) Generate 30 words with each n, starting with "The"
import random

first_word = "The"
length = 30

for n in range(1, 6):
    random.seed(0)
    sample_words = [first_word]
    while len(sample_words) < length:
        # The last n - 1 words, or fewer for the first words
        if n == 1:
            context = ()
        else:
            context = tuple(sample_words[-(n - 1):])
        followers = counts[len(context) + 1][context]
        options, weights = list(followers), list(followers.values())
        next_word = random.choices(options, weights)[0]
        sample_words.append(next_word)
    print(f"n = {n}: " + " ".join(sample_words) + "\n")


# (3) For each n, the fraction of the n-grams of the validation
# sequence never seen in training
for n in range(1, 6):
    unseen = 0
    total = 0
    for i in range(n - 1, len(val_sequence)):
        context = tuple(val_sequence[i - n + 1:i])
        word = val_sequence[i]
        if context not in counts[n] or word not in counts[n][context]:
            unseen += 1
        total += 1
    print(f"n = {n}: {100 * unseen / total:.1f}% never seen")
