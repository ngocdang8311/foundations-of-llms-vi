from datasets import load_dataset

imdb = load_dataset("stanfordnlp/imdb")


# (1) Giữ lại 5.000 bài đánh giá huấn luyện làm tập xác thực và

def clean(texts):
    return [text.replace("<br />", " ") for text in texts]

split = imdb["train"].train_test_split(test_size=5000, seed=0)
train_texts = clean(split["train"]["text"])
val_texts = clean(split["test"]["text"])
test_texts = clean(imdb["test"]["text"])
train_labels = split["train"]["label"]

word_index = {word: int(i)   # từ -> chỉ số, từ 0 đến V - 1

# (1) Mô hình: tầng nhúng từ (thêm một hàng cho từ chưa biết), một
# mạng LSTM đọc chúng theo thứ tự, và một đầu phân loại đọc trạng
# thái cuối cùng. Các bài đánh giá trong cùng một lô có độ dài khác
# nhau: đệm thêm các số 0 để có cùng độ dài và đóng gói kèm độ dài
# thực để LSTM dừng ở từ cuối của mỗi bài mà không đọc phần đệm.
import torch
from torch import nn

device = "cuda" if torch.cuda.is_available() else "cpu"

class LSTMClassifier(nn.Module):
    def __init__(self, V, D, H):
        super().__init__()
        self.embedding = nn.Embedding(V + 1, D)   # + từ chưa biết
        self.lstm = nn.LSTM(D, H, batch_first=True)
        self.head = nn.Linear(H, 1)   # p(positive) = sigmoid

    def forward(self, sequences):
        lengths = torch.tensor([len(s) for s in sequences])
        padded = nn.utils.rnn.pad_sequence(
            [torch.tensor(s) for s in sequences], batch_first=True)
        x = self.embedding(padded.to(device))   # (batch, L, D)
        packed = nn.utils.rnn.pack_padded_sequence(
            x, lengths, batch_first=True, enforce_sorted=False)
        _, (h, c) = self.lstm(packed)   # h: (1, batch, H)
        return self.head(h[0])[:, 0]   # điểm số của mỗi bài đánh giá

torch.manual_seed(0)
model = LSTMClassifier(V, D=100, H=128).to(device)


# (2) Huấn luyện bằng Adam trên các lô 64 bài đánh giá, có cắt
# ngưỡng đạo hàm (gradient clipping). Sau mỗi epoch, đo độ chính
# xác xác thực và giữ lại bản sao của mô hình tốt nhất.
import copy
import random

def predict(model, sequences):
    # p(positive) cho mỗi chuỗi, mỗi lần 500 chuỗi
    probabilities = []
    with torch.no_grad():
        for start in range(0, len(sequences), 500):
            scores = model(sequences[start:start + 500])
            probabilities.append(torch.sigmoid(scores).cpu())
    return torch.cat(probabilities)

def accuracy(model, sequences, labels):
    right = (predict(model, sequences) > 0.5) == torch.tensor(labels)
    return right.float().mean().item()

y_train = torch.tensor(train_labels, dtype=torch.float32)
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
loss_function = nn.BCEWithLogitsLoss()   # sigmoid + cross-entropy
batch_size = 64
best_accuracy = 0
for epoch in range(6):
    order = torch.randperm(len(train_sequences)).tolist()
    for start in range(0, len(order), batch_size):
        batch = order[start:start + batch_size]
        scores = model([train_sequences[i] for i in batch])
        loss = loss_function(scores, y_train[batch].to(device))
        optimizer.zero_grad()
        loss.backward()
        nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        optimizer.step()
    val_accuracy = accuracy(model, val_sequences, val_labels)
    print(f"epoch {epoch + 1}: "
          f"validation accuracy {val_accuracy:.3f}")
    if val_accuracy > best_accuracy:
        best_accuracy, best_epoch = val_accuracy, epoch + 1
        best_model = copy.deepcopy(model)


# (3) Độ chính xác kiểm tra của mô hình tốt nhất, sau đó là bài
# kiểm tra xáo trộn: vẫn các bài đánh giá kiểm tra đó nhưng các từ
# trong mỗi bài bị xáo trộn ngẫu nhiên
test_accuracy = accuracy(best_model, test_sequences, test_labels)
print(f"\ntest accuracy, epoch {best_epoch}: {test_accuracy:.3f}")

random.seed(0)
shuffled_sequences = []
for sequence in test_sequences:
    shuffled = list(sequence)
    random.shuffle(shuffled)
    shuffled_sequences.append(shuffled)
shuffled_accuracy = accuracy(best_model, shuffled_sequences,
                             test_labels)
print(f"test accuracy, words shuffled: {shuffled_accuracy:.3f}")


# (4) Chạy mô hình tốt nhất trên các bài đánh giá khó như ở bài 3.4
hard_sequences = [to_indices(text) for text, label in hard_reviews]
p_positive = predict(best_model, hard_sequences)

names = ["negative", "positive"]
print("\nlabel     p(positive)  right  review")
right_count = 0
for (review, label), p in zip(hard_reviews, p_positive.tolist()):
    right = (p > 0.5) == (label == 1)
    if right:
        right_count += 1
    mark = "yes" if right else "NO"
    short = review if len(review) <= 36 else review[:35] + "…"
    print(f"{names[label]:9} {p:.3f}        {mark:5}  {short}")
print(f"\n{right_count} right out of {len(hard_reviews)}")
