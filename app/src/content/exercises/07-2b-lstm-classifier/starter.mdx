def clean(texts):
    return [text.replace("<br />", " ") for text in texts]

# Từ vựng của bài 3.2: 10.000 từ phổ biến nhất trong các bài đánh
# giá huấn luyện. build_analyzer() trả về hàm mà CountVectorizer
# dùng để tách văn bản thành các từ.
from sklearn.feature_extraction.text import CountVectorizer

vectorizer = CountVectorizer(max_features=10000,
                             token_pattern=r"\b\w+\b")
vectorizer.fit(train_texts)
split_words = vectorizer.build_analyzer()
word_index = {word: int(i)   # từ -> chỉ số, từ 0 đến V - 1
              for word, i in vectorizer.vocabulary_.items()}
V = len(word_index)

# Chuyển mỗi bài đánh giá thành danh sách chỉ số các từ theo đúng
# thứ tự. Các từ ngoài từ vựng nhận chỉ số V. Chỉ giữ lại 400 từ
# cuối cùng đối với các bài đánh giá dài.
max_length = 400

def to_indices(text):
    indices = [word_index.get(word, V) for word in split_words(text)]
    return indices[-max_length:]

train_sequences = [to_indices(text) for text in train_texts]
