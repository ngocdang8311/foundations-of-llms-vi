# The GloVe vectors of Pennington et al., trained on 6 billion
# words of Wikipedia and news (Gigaword), in the copy of
# sentence-transformers on the Hub: 400,000 words in lower case,
# sorted from the most frequent to the least frequent, with 300
# dimensions. Row 0 of the file is a padding token: drop it.
import json

from huggingface_hub import hf_hub_download
from safetensors.torch import load_file

repo = "sentence-transformers/average_word_embeddings_glove.6B.300d"
weights_path = hf_hub_download(
    repo, "0_WordEmbeddings/model.safetensors")
words_path = hf_hub_download(
    repo, "0_WordEmbeddings/whitespacetokenizer_config.json")
glove_vectors = load_file(weights_path)["emb_layer.weight"][1:]
glove_words = json.load(open(words_path))["vocab"][1:]

# Keep the 30,000 most frequent words
words = glove_words[:30000]
vectors = glove_vectors[:30000]
word_index = {word: i for i, word in enumerate(words)}

# The analogy test set of Mikolov et al.: each row reads "Word1 is to
# Word2 as Word3 is to Word4", and Subject is its category
from datasets import load_dataset

analogies = load_dataset("almogtavor/google-analogy-dataset")["train"]

print(f"GloVe: {len(words)} words, vectors of size {vectors.shape[1]}")
print(f"test set: {len(analogies)} analogies")
print(f"example: {analogies[0]}")

# (1) The answer to "a is to b as c is to ?": the word whose vector
# is closest to b - a + c. Divide each vector by its length, so that
# the dot products with the target rank the words by cosine
# similarity. With exclude=True, the three words of the question are
# not allowed as answers.
import math

vectors = vectors / vectors.norm(dim=1, keepdim=True)

def answer(a, b, c, exclude):
    target = (vectors[word_index[b]] - vectors[word_index[a]]
              + vectors[word_index[c]])
    similarities = vectors @ target   # one score per word
    if exclude:
        for word in [a, b, c]:
            similarities[word_index[word]] = -math.inf
    return words[similarities.argmax()]

questions = [("man", "king", "woman"), ("france", "paris", "spain"),
             ("big", "bigger", "small"), ("walk", "walked", "swim"),
             ("man", "doctor", "woman")]
print(f"{'question':34} excluded   allowed")
for a, b, c in questions:
    question = f"{a} is to {b} as {c} is to"
    print(f"{question:34} {answer(a, b, c, True):10} "
          f"{answer(a, b, c, False)}")


# (2) The accuracy on the test set, for the questions whose four
# words are among the 30,000 words. The categories whose name starts
# with "gram" are about grammar (plurals, tenses...), the others
# about meaning (capitals, currencies, family...).
for exclude in [True, False]:
    right = {"meaning": 0, "grammar": 0}
    total = {"meaning": 0, "grammar": 0}
    for row in analogies:
        a, b, c, d = [row[f"Word{i}"].lower() for i in range(1, 5)]
        if not all(word in word_index for word in [a, b, c, d]):
            continue
        kind = "grammar" if row["Subject"].startswith("gram") \
            else "meaning"
        total[kind] += 1
        if answer(a, b, c, exclude) == d:
            right[kind] += 1
    print(f"\nquestion words {'excluded' if exclude else 'allowed'}:")
    for kind in ["meaning", "grammar"]:
        print(f"  {kind}: {right[kind]} right out of {total[kind]} "
              f"({right[kind] / total[kind]:.1%})")
