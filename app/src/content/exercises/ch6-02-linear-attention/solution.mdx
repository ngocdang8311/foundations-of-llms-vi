```python
import torch
import torch.nn.functional as F

torch.manual_seed(0)
B, L, d = 4, 256, 32

# Khởi tạo Q, K, V ngẫu nhiên và hàm ánh xạ đặc trưng dương phi(x)
Q = torch.randn(B, L, d)
K = torch.randn(B, L, d)
V = torch.randn(B, L, d)

def phi(x):
    return F.elu(x) + 1.0

print(f"Q, K, V shape: {tuple(Q.shape)}")
```

```python
# (1) Dạng ma trận song song (Parallel form) khi huấn luyện: O(L^2 d)
def causal_linear_attention_parallel(Q, K, V, eps=1e-6):
    Qp, Kp = phi(Q), phi(K)                        # (B, L, d)
    scores = torch.matmul(Qp, Kp.transpose(-1, -2))  # (B, L, L)
    causal_mask = torch.tril(torch.ones(L, L, dtype=torch.bool))
    scores = scores.masked_fill(~causal_mask, 0.0)
    num = torch.matmul(scores, V)                  # (B, L, d)
    den = scores.sum(dim=-1, keepdim=True) + eps   # (B, L, 1)
    return num / den


# (2) Dạng truy hồi trạng thái (Recurrent form) khi suy luận: O(L d^2)
def causal_linear_attention_recurrent(Q, K, V, eps=1e-6):
    Qp, Kp = phi(Q), phi(K)
    S = torch.zeros(B, d, d)   # Ma trận trạng thái S_t in R^{d x d}
    z = torch.zeros(B, d, 1)   # Vector tổng khóa chuẩn hóa z_t in R^d
    outputs = []

    for t in range(L):
        q_t = Qp[:, t, :].unsqueeze(1)             # (B, 1, d)
        k_t = Kp[:, t, :].unsqueeze(-1)            # (B, d, 1)
        v_t = V[:, t, :].unsqueeze(1)              # (B, 1, d)

        S = S + torch.matmul(k_t, v_t)             # k_t * v_t^T
        z = z + k_t
        num_t = torch.matmul(q_t, S).squeeze(1)    # (B, d)
        den_t = torch.matmul(q_t, z).squeeze(1) + eps
        outputs.append(num_t / den_t)

    return torch.stack(outputs, dim=1), S


y_par = causal_linear_attention_parallel(Q, K, V)
y_rec, S_final = causal_linear_attention_recurrent(Q, K, V)
max_err = (y_par - y_rec).abs().max().item()

print(f"Output shape: {tuple(y_rec.shape)}")
print(f"Trạng thái truy hồi cuối S_L shape: {tuple(S_final.shape)}")
print(f"Sai số tuyệt đối cực đại |Y_parallel - Y_recurrent|: {max_err:.2e}")
print(f"Kiểm chứng trùng khớp (< 1e-6): {max_err < 1e-6}")
```
