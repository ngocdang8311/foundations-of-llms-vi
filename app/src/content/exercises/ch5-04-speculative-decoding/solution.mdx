```python
import torch

torch.manual_seed(2026)
tokens = ["Hà", "Nội", "Sài", "Gòn", "Đà", "Nẵng"]
# Phân phối của mô hình đích lớn p(x) và mô hình nháp nhỏ q(x)
p_target = torch.tensor([0.42, 0.25, 0.15, 0.10, 0.05, 0.03])
q_draft = torch.tensor([0.22, 0.38, 0.18, 0.08, 0.09, 0.05])

print("Tổng p(x):", p_target.sum().item(), "| Tổng q(x):", q_draft.sum().item())
```

```python
def speculative_Verify_samples(p, q, num_trials=200_000):
    # (1) Mô hình nháp đề xuất x ~ q(x)
    x_draft = torch.multinomial(q, num_samples=num_trials, replacement=True)

    # (2) Chấp nhận x với xác suất min(1, p(x) / q(x))
    accept_prob = torch.clamp(p[x_draft] / q[x_draft], max=1.0)
    u = torch.rand(num_trials)
    accepted = u < accept_prob

    # (3) Nếu bị bác bỏ, rút mẫu bù từ p'(x) = norm(max(0, p(x) - q(x)))
    residual = torch.clamp(p - q, min=0.0)
    p_res = residual / residual.sum()
    num_rej = int((~accepted).sum().item())
    x_final = x_draft.clone()
    x_final[~accepted] = torch.multinomial(
        p_res, num_samples=num_rej, replacement=True
    )
    return x_final, accepted.float().mean().item()


alpha_theory = torch.minimum(p_target, q_draft).sum().item()
samples, alpha_emp = speculative_Verify_samples(p_target, q_draft)
counts = torch.bincount(samples, minlength=len(tokens)).float()
p_emp = counts / len(samples)
max_err = (p_emp - p_target).abs().max().item()

print(f"Tỷ lệ chấp nhận lý thuyết α = Σ min(p, q): {alpha_theory*100:.2f}%")
print(f"Tỷ lệ chấp nhận thực nghiệm (200,000 mẫu):  {alpha_emp*100:.2f}%")
print(f"Sai lệch cực đại |p_spec(x) - p(x)|:       {max_err:.4f}\n")
print("Token  |  p_target(x) |  q_draft(x)  |  p_spec(x)")
print("-" * 49)
for i, tok in enumerate(tokens):
    print(
        f"{tok:6s} |     {p_target[i]:.4f}   |"
        f"    {q_draft[i]:.4f}    |    {p_emp[i]:.4f}"
    )
```
