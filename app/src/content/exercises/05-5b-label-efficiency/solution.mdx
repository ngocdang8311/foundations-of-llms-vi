from datasets import load_dataset

imdb = load_dataset("stanfordnlp/imdb")


# (1) Hold out 5,000 training reviews as a validation set, and
# replace the HTML line breaks with spaces
# [Copied from the solution of 3.2, step (1)]

def clean(texts):
    return [text.replace("<br />", " ") for text in texts]

split = imdb["train"].train_test_split(test_size=5000, seed=0)
train_texts = clean(split["train"]["text"])
val_texts = clean(split["test"]["text"])
test_texts = clean(imdb["test"]["text"])
train_labels = split["train"]["label"]
val_labels = split["test"]["label"]
test_labels = imdb["test"]["label"]

print(f"train: {len(train_texts)} reviews")
print(f"validation: {len(val_texts)} reviews")
print(f"test: {len(test_texts)} reviews")


# (2) Count the words of each review, over the 10,000 most
# frequent words of the training reviews. The pattern makes every
# run of letters or digits a word, even a single letter like "a".
# [Copied from the solution of 3.2, step (2)]
from sklearn.feature_extraction.text import CountVectorizer

vectorizer = CountVectorizer(max_features=10000,
                             token_pattern=r"\b\w+\b")
X_train = vectorizer.fit_transform(train_texts)
X_val = vectorizer.transform(val_texts)
X_test = vectorizer.transform(test_texts)


# The GloVe vectors of Pennington et al., trained on 6 billion
# words of Wikipedia and news (Gigaword), in the copy of
# sentence-transformers on the Hub: 400,000 words in lower case,
# sorted from the most frequent to the least frequent, with 300
# dimensions. Row 0 of the file is a padding token: drop it.
# [Copied from the starter of 5.4]
import json

from huggingface_hub import hf_hub_download
from safetensors.torch import load_file

repo = "sentence-transformers/average_word_embeddings_glove.6B.300d"
weights_path = hf_hub_download(
    repo, "0_WordEmbeddings/model.safetensors")
words_path = hf_hub_download(
    repo, "0_WordEmbeddings/whitespacetokenizer_config.json")
glove_vectors = load_file(weights_path)["emb_layer.weight"][1:]
glove_words = json.load(open(words_path))["vocab"][1:]


# (1) The pre-trained embedding matrix: column j is the GloVe vector
# of word j of the vocabulary, in the order of the count vectors.
# The few words that GloVe does not know keep a column of zeros.
# [Copied from the solution of 5.5a, step (1)]
import torch

glove_index = {word: i for i, word in enumerate(glove_words)}
vocabulary = vectorizer.get_feature_names_out()
D = glove_vectors.shape[1]   # 300
W_glove = torch.zeros(D, len(vocabulary))
missing = []
for j, word in enumerate(vocabulary):
    if word in glove_index:
        W_glove[:, j] = glove_vectors[glove_index[word]]
    else:
        missing.append(word)
print(f"{len(missing)} words without a GloVe vector: {missing[:5]}")


# (2) The deep averaging network of exercise 4.3, with D = 300. Its
# first layer is W: random (from scratch), or the GloVe vectors,
# kept fixed (frozen) or trained with the rest (fine-tuned).
# [Copied from the solution of 5.5a, step (2)]
from torch import nn

def frequencies(X):
    counts = torch.tensor(X.toarray(), dtype=torch.float32)
    lengths = counts.sum(dim=1, keepdim=True)
    return counts / lengths

def make_model(embeddings):
    torch.manual_seed(0)
    model = nn.Sequential(
        nn.Linear(len(vocabulary), D, bias=False),   # average
        nn.Linear(D, 100),
        nn.ReLU(),
        nn.Linear(100, 100),
        nn.ReLU(),
        nn.Linear(100, 1),   # p(positive) = sigmoid(score)
    )
    if embeddings != "from scratch":
        model[0].weight.data = W_glove.clone()
    if embeddings == "frozen":
        model[0].weight.requires_grad_(False)   # no gradient
    return model


# (3) A function that trains a model for 1,500 batches of 64
# reviews, drawn at random from the training reviews. Every 100
# batches, it measures the validation accuracy, and keeps a copy of
# the best model.
# [Copied from the solution of 5.5a, step (3)]
import copy

def predict(model, X):
    # p(positive) for each review of X, 1,000 reviews at a time
    probabilities = []
    with torch.no_grad():
        for start in range(0, X.shape[0], 1000):
            scores = model(frequencies(X[start:start + 1000]))
            probabilities.append(torch.sigmoid(scores[:, 0]))
    return torch.cat(probabilities)

def accuracy(model, X, labels):
    right = (predict(model, X) > 0.5) == torch.tensor(labels)
    return right.float().mean().item()

def train(model, X, labels, n_batches=1500):
    y = torch.tensor(labels, dtype=torch.float32)
    trained = [p for p in model.parameters() if p.requires_grad]
    optimizer = torch.optim.Adam(trained, lr=1e-3)
    loss_function = nn.BCEWithLogitsLoss()
    best_accuracy = 0
    for step in range(1, n_batches + 1):
        batch = torch.randint(len(y), (64,)).tolist()
        scores = model(frequencies(X[batch]))[:, 0]
        loss = loss_function(scores, y[batch])
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
        if step % 100 == 0:
            val_accuracy = accuracy(model, X_val, val_labels)
            if val_accuracy > best_accuracy:
                best_accuracy = val_accuracy
                best_model = copy.deepcopy(model)
    return best_model

# (1) For each number of labels, train the three versions on random
# subsets of the training reviews, drawn with random.sample: 3
# subsets for the small sizes, whose accuracy varies a lot from one
# subset to the next, and 1 for the others. Average the test
# accuracies over the subsets.
import random

sizes = [100, 300, 1000, 3000, 10000, 20000]
versions = ["from scratch", "frozen", "fine-tuned"]
results = {version: [] for version in versions}
for n in sizes:
    n_subsets = 3 if n <= 1000 else 1
    for version in versions:
        total = 0
        for seed in range(n_subsets):
            random.seed(seed)
            chosen = random.sample(range(len(train_labels)), n)
            labels = [train_labels[i] for i in chosen]
            model = train(make_model(version), X_train[chosen], labels)
            total += accuracy(model, X_test, test_labels)
        results[version].append(total / n_subsets)
        print(f"{n:6} labels, {version:12}  "
              f"test accuracy {results[version][-1]:.3f}")


# (2) Plot the test accuracy against the number of labels
import matplotlib.pyplot as plt

plt.figure(figsize=(5, 3.5))
for version in versions:
    plt.plot(sizes, results[version], "o-", label=version)
plt.xscale("log")
plt.xlabel("number of labeled reviews")
plt.ylabel("test accuracy")
plt.legend()
plt.show()
