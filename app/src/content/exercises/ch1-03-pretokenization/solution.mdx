```python
import re

corpus = """Beautiful is better than ugly. Explicit is better than implicit.
Simple is better than complex. Complex is better than complicated.
Special cases aren't special enough to break the rules.
Although practicality beats purity, errors should never pass silently.
Now is better than never, although never is often better than right now.
If the implementation is hard to explain, it's a bad idea!
Language models do not read raw text directly as words or characters.
Instead, a tokenizer converts text into a sequence of integer token IDs.
Common words like model, models, model., model!, model? share the root model.
When training on numbers like 123456 or 2024, GPT-4 splits digits into
groups of at most 3 digits so arithmetic is more consistent across tokens.
Subword tokenization helps neural networks generalize across different forms
of the same root word, such as train, trained, trainer, and training,
or token, tokens, tokenizer, and tokenization in language modeling."""

# Biểu thức chính quy pre-tokenization rút gọn theo chuẩn GPT-4 (cl100k_base)
GPT4_PAT = re.compile(
    r"'(?:[sdmt]|ll|ve|re)| ?[A-Za-z]+| ?\d{1,3}| ?[^\sA-Za-z\d]+|\s+"
)
print("Tách câu thử nghiệm:",
      GPT4_PAT.findall("It's 123456: model., model!"))
```

```python
# (1) Tách văn bản thành danh sách các mảnh độc lập bằng Regex
chunks = GPT4_PAT.findall(corpus)
print(f"Tổng số mảnh pre-tokenized: {len(chunks)}")


# (2) Huấn luyện BPE chỉ đếm và gộp cặp bên trong từng mảnh (chunk)
def train_bpe(chunks_list, num_merges=80):
    seqs = [list(c.encode("utf-8")) for c in chunks_list]
    vocab = {i: bytes([i]) for i in range(256)}
    for step in range(num_merges):
        counts = {}
        for seq in seqs:
            for pair in zip(seq, seq[1:]):
                counts[pair] = counts.get(pair, 0) + 1
        best_pair = max(counts, key=counts.get)
        new_id = 256 + step
        vocab[new_id] = vocab[best_pair[0]] + vocab[best_pair[1]]
        new_seqs = []
        for seq in seqs:
            merged, i = [], 0
            while i < len(seq):
                if i < len(seq) - 1 and (seq[i], seq[i + 1]) == best_pair:
                    merged.append(new_id)
                    i += 2
                else:
                    merged.append(seq[i])
                    i += 1
            new_seqs.append(merged)
        seqs = new_seqs
    return [
        vocab[256 + s].decode("utf-8", errors="replace")
        for s in range(num_merges)
    ]


# (3) So sánh BPE ngây thơ (1 chuỗi liền) và BPE có Regex pre-tokenizer
naive_tokens = train_bpe([corpus], num_merges=80)
regex_tokens = train_bpe(chunks, num_merges=80)


def is_cross_boundary(t):
    punct_alpha = any(p in t for p in ",.!?") and any(c.isalpha() for c in t)
    multi_words = len(t.strip().split()) > 1
    return punct_alpha or multi_words


bad_naive = [t for t in naive_tokens if is_cross_boundary(t)]
bad_regex = [t for t in regex_tokens if is_cross_boundary(t)]

print("\nToken gộp xuyên ranh giới từ/dấu câu:")
print(f"  BPE ngây thơ ({len(bad_naive)}/80): {bad_naive[:8]}")
print(f"  BPE có Regex ({len(bad_regex)}/80): {bad_regex}")
top_regex = sorted(regex_tokens, key=len, reverse=True)[:10]
print("10 token dài nhất (BPE có Regex):", top_regex)
```
