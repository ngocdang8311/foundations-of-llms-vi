from datasets import load_dataset

wikitext = load_dataset("Salesforce/wikitext", "wikitext-2-v1")


# (1) Tạo chuỗi huấn luyện bằng cách tách mỗi dòng theo khoảng trắng
# [Sao chép từ lời giải bài 2.1, bước (1)]
train_sequence = []
for line in wikitext["train"]["text"]:
    train_sequence.extend(line.split())

print(f"text: {len(train_sequence)} words")
print(f"vocabulary: {len(set(train_sequence))} words")


# Chuỗi xác thực và chuỗi kiểm tra, được tạo theo cùng cách
# [Sao chép từ phần khởi đầu bài 2.2]
val_sequence = []
for line in wikitext["validation"]["text"]:
    val_sequence.extend(line.split())

test_sequence = []
for line in wikitext["test"]["text"]:
    test_sequence.extend(line.split())

print(f"validation: {len(val_sequence)} words")
print(f"test: {len(test_sequence)} words")


# Gán một chỉ số cho mỗi từ trong từ vựng
# [Sao chép từ phần khởi đầu bài 4.2]
vocabulary = sorted(set(train_sequence))
word_index = {word: i for i, word in enumerate(vocabulary)}


# ----------------------
#  Mới trong bài tập này
# ----------------------
# Cắt chuỗi thành n luồng chỉ số: hàng i chứa phần thứ i của văn
# bản. Mỗi lô đọc n luồng song song cạnh nhau để mỗi luồng được
# nối tiếp từ lô này sang lô tiếp theo.
import torch

def make_streams(sequence, n):
    indices = torch.tensor([word_index[word] for word in sequence])
    length = len(indices) // n   # bỏ một vài từ cuối cùng
    return indices[:n * length].view(n, length)

train_streams = make_streams(train_sequence, 32)
val_streams = make_streams(val_sequence, 10)
test_streams = make_streams(test_sequence, 10)

print(f"\ntrain: {tuple(train_streams.shape)} (streams, words)")
first = [vocabulary[i] for i in train_streams[1, :8].tolist()]
print(f"start of stream 1: {' '.join(first)}")
