from datasets import load_dataset

wikitext = load_dataset("Salesforce/wikitext", "wikitext-2-v1")


# (1) Tạo chuỗi huấn luyện bằng cách tách mỗi dòng theo khoảng trắng
# [Sao chép từ lời giải bài 2.1, bước (1)]
train_sequence = []
for line in wikitext["train"]["text"]:
    train_sequence.extend(line.split())

print(f"text: {len(train_sequence)} words")
print(f"vocabulary: {len(set(train_sequence))} words")


# Chuỗi xác thực và chuỗi kiểm tra, được tạo theo cùng cách
# [Sao chép từ phần khởi đầu bài 2.2]
val_sequence = []
for line in wikitext["validation"]["text"]:
    val_sequence.extend(line.split())

test_sequence = []
for line in wikitext["test"]["text"]:
    test_sequence.extend(line.split())

print(f"validation: {len(val_sequence)} words")
print(f"test: {len(test_sequence)} words")


# Gán một chỉ số cho mỗi từ trong từ vựng
# [Sao chép từ phần khởi đầu bài 4.2]
vocabulary = sorted(set(train_sequence))
word_index = {word: i for i, word in enumerate(vocabulary)}


# ----------------------
#  Mới trong bài tập này

# Cắt chuỗi thành n luồng chỉ số: hàng i chứa phần thứ i của văn
# bản. Mỗi lô đọc n luồng song song cạnh nhau để mỗi luồng được
# nối tiếp từ lô này sang lô tiếp theo.
import torch

def make_streams(sequence, n):
    indices = torch.tensor([word_index[word] for word in sequence])
    length = len(indices) // n   # bỏ một vài từ cuối cùng
    return indices[:n * length].view(n, length)

train_streams = make_streams(train_sequence, 32)
val_streams = make_streams(val_sequence, 10)
test_streams = make_streams(test_sequence, 10)

print(f"\ntrain: {tuple(train_streams.shape)} (streams, words)")
first = [vocabulary[i] for i in train_streams[1, :8].tolist()]

# (1) Mô hình: tầng nhúng, mạng LSTM và điểm số cho từng từ trong
# từ vựng tại mỗi vị trí. forward() nhận trạng thái từ đoạn trước
# để lại và trả về trạng thái mới. Khởi tạo các vector nhúng nhỏ
# như ở bài 4.2.
from torch import nn
import torch.nn.functional as F

class LSTMLanguageModel(nn.Module):
    def __init__(self, V, D, H):
        super().__init__()
        self.embedding = nn.Embedding(V, D)
        self.lstm = nn.LSTM(D, H, batch_first=True)
        self.output = nn.Linear(H, V)

    def forward(self, x, state=None):
        e = self.embedding(x)            # (batch, steps, D)
        h, state = self.lstm(e, state)   # (batch, steps, H)
        return self.output(h), state     # (batch, steps, V)

device = "cuda" if torch.cuda.is_available() else "cpu"
V = len(vocabulary)
torch.manual_seed(0)
model = LSTMLanguageModel(V, D=256, H=256).to(device)
nn.init.normal_(model.embedding.weight, std=0.1)


# (2) Các đoạn dài 35 từ: x chứa các từ của đoạn, y chứa các từ
# theo sau chúng ở vị trí kế tiếp
steps = 35

def chunks(streams):
    pairs = []
    for start in range(0, streams.shape[1] - 1, steps):
        y = streams[:, start + 1:start + 1 + steps]
        x = streams[:, start:start + y.shape[1]]
        pairs.append((x.to(device), y.to(device)))
    return pairs


# (3) Độ hỗn loạn (perplexity) trên một tập luồng, đọc nối tiếp
# từng đoạn và truyền trạng thái qua các đoạn
import math

def perplexity(model, streams):
    total_loss, count, state = 0, 0, None
    with torch.no_grad():
        for x, y in chunks(streams):
            scores, state = model(x, state)
            total_loss += F.cross_entropy(scores.reshape(-1, V),
                                          y.reshape(-1),
                                          reduction="sum").item()
            count += y.numel()
    return math.exp(total_loss / count)


# (4) Huấn luyện với Adam và cắt ngưỡng đạo hàm. Trạng thái được
# truyền từ đoạn này sang đoạn tiếp theo, nhưng detach() cắt liên
# kết với các đoạn trước để đạo hàm dừng lại ở đoạn hiện tại.
import copy

optimizer = torch.optim.Adam(model.parameters(), lr=2e-3)
best_perplexity = math.inf
for epoch in range(3):
    state = None
    for x, y in chunks(train_streams):
        if state is not None:
            state = (state[0].detach(), state[1].detach())
        scores, state = model(x, state)
        loss = F.cross_entropy(scores.reshape(-1, V), y.reshape(-1))
        optimizer.zero_grad()
        loss.backward()
        nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        optimizer.step()
    val_perplexity = perplexity(model, val_streams)
    print(f"epoch {epoch + 1}: validation perplexity "
          f"{val_perplexity:.1f}")
    if val_perplexity < best_perplexity:
        best_perplexity, best_epoch = val_perplexity, epoch + 1
        best_model = copy.deepcopy(model)


# (5) Độ hỗn loạn trên tập kiểm tra của mô hình tốt nhất
test_perplexity = perplexity(best_model, test_streams)
print(f"\ntest perplexity, epoch {best_epoch}: {test_perplexity:.1f}")


# (6) Sinh 50 từ, từng từ một: trạng thái mang thông tin của toàn
# bộ các từ đã sinh cho đến thời điểm hiện tại
first_word = "The"
length = 50
torch.manual_seed(0)
sample_words = [first_word]
x = torch.tensor([[word_index[first_word]]]).to(device)
state = None
with torch.no_grad():
    for _ in range(length - 1):
        scores, state = best_model(x, state)
        probabilities = torch.softmax(scores[0, -1], dim=0)
        next_index = torch.multinomial(probabilities, 1).item()
        sample_words.append(vocabulary[next_index])
        x = torch.tensor([[next_index]]).to(device)
print(f"\n{' '.join(sample_words)}")
