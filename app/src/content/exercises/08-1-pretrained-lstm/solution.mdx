# mạng LSTM đọc chúng theo thứ tự, và một đầu phân loại đọc trạng

# nhau: đệm thêm các số 0 để có cùng độ dài và đóng gói kèm độ dài
# thực để LSTM dừng ở từ cuối của mỗi bài mà không đọc phần đệm.
# [Sao chép từ lời giải bài 7.2b, bước (1)]
import torch
from torch import nn

device = "cuda" if torch.cuda.is_available() else "cpu"

class LSTMClassifier(nn.Module):
    def __init__(self, V, D, H):
        super().__init__()
        self.embedding = nn.Embedding(V + 1, D)   # + từ chưa biết
        self.lstm = nn.LSTM(D, H, batch_first=True)
        self.head = nn.Linear(H, 1)   # p(positive) = sigmoid

    def forward(self, sequences):
        lengths = torch.tensor([len(s) for s in sequences])
        padded = nn.utils.rnn.pad_sequence(
            [torch.tensor(s) for s in sequences], batch_first=True)
        x = self.embedding(padded.to(device))   # (batch, L, D)
        packed = nn.utils.rnn.pack_padded_sequence(
            x, lengths, batch_first=True, enforce_sorted=False)
        _, (h, c) = self.lstm(packed)   # h: (1, batch, H)
        return self.head(h[0])[:, 0]   # điểm số của mỗi bài đánh giá

torch.manual_seed(0)

# (1) Mô hình ngôn ngữ: tầng nhúng và mạng LSTM của bộ phân loại ở
# bài 7.2b, cùng tầng đầu ra chấm điểm cho từng từ trong từ vựng tại
# mỗi vị trí như ở bài 7.3. Các từ chưa biết (chỉ số V) là một từ
# nữa mà mô hình có thể dự đoán. Khởi tạo nhúng nhỏ như bài 4.2.
import torch.nn.functional as F

class LSTMLanguageModel(nn.Module):
    def __init__(self, V, D, H):
        super().__init__()
        self.embedding = nn.Embedding(V + 1, D)   # + từ chưa biết
        self.lstm = nn.LSTM(D, H, batch_first=True)
        self.output = nn.Linear(H, V + 1)

    def forward(self, x, state=None):
        e = self.embedding(x)            # (batch, steps, D)
        h, state = self.lstm(e, state)   # (batch, steps, H)
        return self.output(h), state     # (batch, steps, V + 1)

torch.manual_seed(0)
language_model = LSTMLanguageModel(V, D=100, H=128).to(device)
nn.init.normal_(language_model.embedding.weight, std=0.1)


# (2) Một lô bài đánh giá: đầu vào x gồm các từ của mỗi bài trừ từ
# cuối, nhãn y gồm các từ theo sau chúng. pad_sequence đệm vào cuối
# các bài ngắn: LSTM đọc từ trái sang phải nên phần đệm không làm
# đổi trạng thái của các từ thật. Nhãn được đệm bằng -100, giá trị
# mà F.cross_entropy bỏ qua nên phần đệm không tính vào hàm mất mát.
from torch.nn.utils.rnn import pad_sequence

def make_batch(sequences):
    x = pad_sequence([torch.tensor(s[:-1]) for s in sequences],
                     batch_first=True)
    y = pad_sequence([torch.tensor(s[1:]) for s in sequences],
                     batch_first=True, padding_value=-100)
    return x.to(device), y.to(device)


# (3) Độ hỗn loạn trên danh sách các bài đánh giá, mỗi lần 50 bài
import math

def perplexity(model, sequences):
    total_loss, count = 0, 0
    with torch.no_grad():
        for start in range(0, len(sequences), 50):
            x, y = make_batch(sequences[start:start + 50])
            scores, _ = model(x)
            total_loss += F.cross_entropy(scores.reshape(-1, V + 1),
                                          y.reshape(-1),
                                          reduction="sum").item()
            count += (y != -100).sum().item()
    return math.exp(total_loss / count)


# (4) Huấn luyện một epoch trên các bài đánh giá không nhãn theo lô
# 64 bài, với Adam và cắt ngưỡng đạo hàm như ở bài 7.3
optimizer = torch.optim.Adam(language_model.parameters(), lr=2e-3)
order = torch.randperm(len(unlabeled_sequences)).tolist()
for start in range(0, len(order), 64):
    batch = [unlabeled_sequences[i] for i in order[start:start + 64]]
    x, y = make_batch(batch)
    scores, _ = language_model(x)
    loss = F.cross_entropy(scores.reshape(-1, V + 1), y.reshape(-1))
    optimizer.zero_grad()
    loss.backward()
    nn.utils.clip_grad_norm_(language_model.parameters(), 1.0)
    optimizer.step()

val_perplexity = perplexity(language_model, val_sequences)
print(f"validation perplexity: {val_perplexity:.1f}")

        probabilities = torch.softmax(scores[0, -1], dim=0)
        next_index = torch.multinomial(probabilities, 1).item()
        sample_words.append(vocabulary[next_index])
        x = torch.tensor([[next_index]]).to(device)
print(f"\n{' '.join(sample_words)}")

# (1) Mô hình ngôn ngữ biết gì về sắc thái cảm xúc? Đóng băng nó và
# huấn luyện hồi quy logistic như ở bài 3.2 trên trạng thái sau từ
# cuối cùng của mỗi bài đánh giá. Làm tương tự với chính mạng đó
# trước khi huấn luyện để so sánh.
from sklearn.linear_model import LogisticRegression

def last_states(model, sequences):
    # Trạng thái sau từ cuối cùng của mỗi bài đánh giá, 500 bài/lần
    states = []
    with torch.no_grad():
        for start in range(0, len(sequences), 500):
            batch = sequences[start:start + 500]
            lengths = torch.tensor([len(s) for s in batch])
            padded = pad_sequence([torch.tensor(s) for s in batch],
                                  batch_first=True)
            e = model.embedding(padded.to(device))
            h, _ = model.lstm(e)   # (batch, L, H): sau mỗi từ
            last = h[torch.arange(len(batch)), lengths - 1]
            states.append(last.cpu())
    return torch.cat(states)

torch.manual_seed(0)
untrained_model = LSTMLanguageModel(V, D=100, H=128).to(device)
nn.init.normal_(untrained_model.embedding.weight, std=0.1)

for name, lm in [("before training", untrained_model),
                 ("after training", language_model)]:
    probe = LogisticRegression(max_iter=1000)
    probe.fit(last_states(lm, train_sequences), train_labels)
    probe_accuracy = probe.score(last_states(lm, test_sequences),
                                 test_labels)
    print(f"LSTM {name}: test accuracy {probe_accuracy:.3f}")


# (2) Tinh chỉnh bộ phân loại của bài 7.2b trên 1.000 bài đánh giá
# có nhãn với ba cách khởi tạo: ngẫu nhiên (từ đầu), lấy tầng nhúng
# của mô hình ngôn ngữ, hoặc lấy cả tầng nhúng lẫn mạng LSTM của nó.
# Đầu phân loại đều tạo mới trong cả ba cách. Như ở bài 5.5b, lấy
# ngẫu nhiên các lô từ tập có nhãn, đo độ chính xác trên 1.000 bài
# xác thực sau mỗi 50 lô và giữ lại mô hình tốt nhất.
import copy
import random

def predict(model, sequences):
    # p(positive) cho mỗi chuỗi, mỗi lần 500 chuỗi
    probabilities = []
    with torch.no_grad():
        for start in range(0, len(sequences), 500):
            scores = model(sequences[start:start + 500])
            probabilities.append(torch.sigmoid(scores).cpu())
    return torch.cat(probabilities)

def accuracy(model, sequences, labels):
    right = (predict(model, sequences) > 0.5) == torch.tensor(labels)
    return right.float().mean().item()

def fine_tune(start, chosen, n_batches=400):
    torch.manual_seed(0)
    model = LSTMClassifier(V, D=100, H=128).to(device)
    lm = language_model
    if start != "from scratch":
        model.embedding.load_state_dict(lm.embedding.state_dict())
    if start == "embeddings and LSTM":
        model.lstm.load_state_dict(lm.lstm.state_dict())
    y = torch.tensor(train_labels, dtype=torch.float32)
    optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
    loss_function = nn.BCEWithLogitsLoss()
    random.seed(0)   # dùng cùng các lô cho cả ba cách khởi tạo
    best_accuracy = 0
    for step in range(1, n_batches + 1):
        batch = random.choices(chosen, k=32)
        scores = model([train_sequences[i] for i in batch])
        loss = loss_function(scores, y[batch].to(device))
        optimizer.zero_grad()
        loss.backward()
        nn.utils.clip_grad_norm_(model.parameters(), 1.0)
        optimizer.step()
        if step % 50 == 0:
            val_accuracy = accuracy(model, val_sequences[:1000],
                                    val_labels[:1000])
            if val_accuracy > best_accuracy:
                best_accuracy = val_accuracy
                best_model = copy.deepcopy(model)
    return best_model


# (3) Tinh chỉnh mỗi cách khởi tạo trên 3 tập ngẫu nhiên gồm 1.000
# bài đánh giá có nhãn và lấy trung bình độ chính xác kiểm tra
starts = ["from scratch", "embeddings", "embeddings and LSTM"]
results = {start: [] for start in starts}
for seed in range(3):
    random.seed(seed)
    chosen = random.sample(range(len(train_labels)), 1000)
    for start in starts:
        model = fine_tune(start, chosen)
        results[start].append(accuracy(model, test_sequences,
                                       test_labels))

print("\nfine-tuned on 1,000 labeled reviews:")
for start in starts:
    mean_accuracy = sum(results[start]) / len(results[start])
    print(f"  {start:20} test accuracy {mean_accuracy:.3f}")
