```python
import torch

# A sequence of L random embeddings of size D, as a batch of one
# sequence: PyTorch's recurrent layers with batch_first=True expect
# a tensor of shape (batch, L, D)
L, D = 100, 100
torch.manual_seed(0)
embeddings = torch.randn(1, L, D)

print(f"embeddings: {tuple(embeddings.shape)}")
```

```python
# (1) A plain RNN and an LSTM with hidden states of size H, untrained,
# with the default initialization of PyTorch
from torch import nn

H = 100
torch.manual_seed(0)
rnn = nn.RNN(D, H, batch_first=True)
lstm = nn.LSTM(D, H, batch_first=True)


# (2) The norm of the gradient of the last state with respect to each
# embedding. backward() needs a single number: take the sum of the
# components of the last state.
def gradient_norms(model, embeddings):
    x = embeddings.clone().requires_grad_()   # track the gradient
    states, _ = model(x)   # (1, L, H): the state after each word
    states[0, -1].sum().backward()
    # One norm per word, computed in float64: in float32, the
    # squares of the tiny components would round to 0
    return x.grad[0].double().norm(dim=1)


# (3) A copy of the LSTM with the bias of its forget gate set to 3.
# PyTorch adds two biases, bias_ih_l0 and bias_hh_l0, each holding
# the biases of the 4 gates one after the other: input, forget,
# cell, output, H values each.
import copy

open_lstm = copy.deepcopy(lstm)
with torch.no_grad():
    open_lstm.bias_ih_l0[H:2 * H] = 3.0   # the forget gate
    open_lstm.bias_hh_l0[H:2 * H] = 0.0


# (4) Plot the norms against the distance from the end of the
# sequence: 0 for the last embedding, L - 1 for the first one
import matplotlib.pyplot as plt

models = {"RNN": rnn, "LSTM": lstm,
          "LSTM, forget bias 3": open_lstm}
distances = list(range(L))
plt.figure(figsize=(5, 3.5))
for name, model in models.items():
    norms = gradient_norms(model, embeddings).flip(0).tolist()
    plt.plot(distances, norms, label=name)
    print(f"{name:20}  distance 10: {norms[10]:.1e}   "
          f"50: {norms[50]:.1e}   99: {norms[99]:.1e}")
plt.yscale("log")
plt.xlabel("distance from the end of the sequence")
plt.ylabel("norm of the gradient")
plt.legend()
plt.show()
```
