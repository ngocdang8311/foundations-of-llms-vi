```python
import torch

# Một chuỗi gồm L vector nhúng ngẫu nhiên kích thước D, dưới dạng
# lô gồm một chuỗi: các tầng hồi quy của PyTorch với batch_first=True
# nhận đầu vào là tensor có kích thước (batch, L, D)
L, D = 100, 100
torch.manual_seed(0)
embeddings = torch.randn(1, L, D)

print(f"embeddings: {tuple(embeddings.shape)}")
```

```python
# (1) Một mạng RNN thuần và một mạng LSTM với trạng thái ẩn kích
# thước H, chưa huấn luyện, dùng khởi tạo mặc định của PyTorch
from torch import nn

H = 100
torch.manual_seed(0)
rnn = nn.RNN(D, H, batch_first=True)
lstm = nn.LSTM(D, H, batch_first=True)


# (2) Chuẩn đạo hàm của trạng thái cuối cùng theo từng vector nhúng.
# backward() cần một số vô hướng: lấy tổng các thành phần của trạng
# thái cuối cùng.
def gradient_norms(model, embeddings):
    x = embeddings.clone().requires_grad_()   # theo dõi đạo hàm
    states, _ = model(x)   # (1, L, H): trạng thái sau mỗi từ
    states[0, -1].sum().backward()
    # Một chuẩn cho mỗi từ, tính ở float64: trong float32, bình
    # phương của các thành phần cực nhỏ sẽ bị làm tròn thành 0
    return x.grad[0].double().norm(dim=1)


# (3) Một bản sao của LSTM với độ chệch (bias) của cổng quên đặt
# bằng 3. PyTorch cộng hai vector độ chệch, bias_ih_l0 và bias_hh_l0,
# mỗi vector chứa độ chệch của 4 cổng nối tiếp nhau: cổng vào, cổng
# quên, ô nhớ, cổng ra, mỗi cổng H giá trị.
import copy

open_lstm = copy.deepcopy(lstm)
with torch.no_grad():
    open_lstm.bias_ih_l0[H:2 * H] = 3.0   # cổng quên
    open_lstm.bias_hh_l0[H:2 * H] = 0.0


# (4) Vẽ biểu đồ các chuẩn theo khoảng cách tính từ cuối chuỗi:
# 0 cho vector nhúng cuối cùng, L - 1 cho vector nhúng đầu tiên
import matplotlib.pyplot as plt

models = {"RNN": rnn, "LSTM": lstm,
          "LSTM, forget bias 3": open_lstm}
distances = list(range(L))
plt.figure(figsize=(5, 3.5))
for name, model in models.items():
    norms = gradient_norms(model, embeddings).flip(0).tolist()
    plt.plot(distances, norms, label=name)
    print(f"{name:20}  distance 10: {norms[10]:.1e}   "
          f"50: {norms[50]:.1e}   99: {norms[99]:.1e}")
plt.yscale("log")
plt.xlabel("distance from the end of the sequence")
plt.ylabel("norm of the gradient")
plt.legend()
plt.show()
```
